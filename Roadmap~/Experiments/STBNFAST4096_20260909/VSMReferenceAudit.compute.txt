#pragma kernel VSMReferenceAudit
#pragma use_dxc
#pragma require Native16Bit
#define VIVID_SMRT_REFERENCE
#include "Packages/com.vivid.render-pipelines/Shaders/Core/Private/CSMShadowResolve.compute"
RWTexture2DArray<float> _ReferenceAccumulator;
int _ReferencePhase, _ReferenceBatch, _ReferenceReset;
float4 _ReferenceOriginSize;
[numthreads(8,8,1)]
void VSMReferenceAudit(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_ReferenceOriginSize.z || id.y >= (uint)_ReferenceOriginSize.w) return;
    uint2 pixel = id.xy + (uint2)_ReferenceOriginSize.xy;
    float depth = _DepthTexture.Load(int3(pixel,0));
    float sum = 0;
    if (IsSkyPixel(depth)) sum = 32;
    else
    {
        float3 position = ReconstructWorldPosition(pixel,depth);
        float3 normal = DecodeVividNormalOct(_GBuffer1.Load(int3(pixel,0)).xy);
        normal = ReconstructVSMReceiverNormal(pixel,depth,position,normal);
        [loop] for(uint sampleIndex=0;sampleIndex<32;sampleIndex++)
        {
            g_VSMReferenceSample = (uint)_ReferenceBatch * 32u + sampleIndex;
            sum += ResolveVSMReceiver(position,normal,pixel);
        }
    }
    uint3 coord=uint3(id.xy,(uint)_ReferencePhase);
    _ReferenceAccumulator[coord] = (_ReferenceReset != 0 ? 0 : _ReferenceAccumulator[coord]) + sum / 32.0;
}
