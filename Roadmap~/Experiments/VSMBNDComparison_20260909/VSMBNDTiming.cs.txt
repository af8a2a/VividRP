using System;
using System.IO;
using System.Reflection;
using System.Collections.Generic;
using Unity.Profiling;
using UnityEditor;
using UnityEngine;
using VividRP.Runtime;
using VividRP.Runtime.RenderPass.Core;
namespace VividRP.Editor {
internal static class VSMBNDTiming {
 static VSMBaselineSession session;
 static double started;
 static int stage, originalLimit;
 static string root;
 static VividAdditionalLightData light;
 static float originalAngle;
 static readonly string[] Labels={"legacy4_a","bnd4_a","legacy8_a","bnd8_a","bnd8_b","legacy8_b","bnd4_b","legacy4_b"};
 static readonly int[] Limits={256,256,256,256,256,256,256,256};
 static ProfilerRecorder recorder;
 static readonly List<double> scopedSamples=new List<double>(4096);
 static int lastFrame;
 [MenuItem("Tools/VividRP/Diagnostics/Run Temporary VSM BND Timing")]
 static void Run(){
  if(Application.dataPath!="E:/VividRP_Reborn/Assets"||!EditorApplication.isPlaying||session!=null)throw new InvalidOperationException("Use project Play Mode.");
  originalLimit=VirtualShadowMapPrototypeRuntime.EditorPhysicalPageLimit;
  foreach(var l in UnityEngine.Object.FindObjectsByType<Light>(FindObjectsSortMode.None))if(l.type==LightType.Directional){light=l.GetComponent<VividAdditionalLightData>();break;}
  if(light==null)throw new InvalidOperationException("No sun");originalAngle=light.angularDiameter;light.angularDiameter=7.1f;
  root="Packages/VividRP/Temp~/vsm-bnd-timing/"+DateTime.UtcNow.ToString("yyyyMMdd_HHmmss_fff");Directory.CreateDirectory(root);stage=0;
  CSMShadowResolvePass.EditorBNDTimingCamera=EntityId.ToULong(Camera.main.GetEntityId());
  recorder=ProfilerRecorder.StartNew(ProfilerCategory.Render,"VSM.BNDComparison.MainCameraResolve",1,
   ProfilerRecorderOptions.StartImmediately|ProfilerRecorderOptions.WrapAroundWhenCapacityReached|ProfilerRecorderOptions.SumAllSamplesInFrame|ProfilerRecorderOptions.GpuRecorder);
  try{Begin();}catch{Finish();throw;}
  EditorApplication.update+=Tick;AssemblyReloadEvents.beforeAssemblyReload+=Finish;
 }
 static void Begin(){
  scopedSamples.Clear();lastFrame=-1;
  CSMShadowResolvePass.EditorBNDLegacyPhase=Labels[stage].StartsWith("legacy",StringComparison.Ordinal)?1:0;
  CSMShadowResolvePass.EditorBNDSamplingFrame=-1;
  VirtualShadowMapPrototypeRuntime.EditorPhysicalPageLimit=Limits[stage];VirtualShadowMapPrototypeRuntime.ReleaseResources();
  var c=new VSMBaselineCase{name=Labels[stage],resolution=4096,firstLevel=0,screenDensity=true,pcf=true,transition=.2f,maxDistance=150,screenSpaceDenoise=true};
  session=new VSMBaselineSession(Camera.main,new[]{c},5,5,120,4096,root+"/"+Labels[stage],"4096 production 256 pages; source angle7.1; BND phase only; main-camera Resolve GPU scope plus whole-frame indicative timers",false);
  // Cold diagnostic setup only. Keep shipping timing classes unchanged.
  var settings=(CascadedShadowSettingsVolume)typeof(VSMBaselineSession).GetField("m_Settings",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(session);
  settings.virtualShadowMapSMRT.Override(true);
  settings.virtualShadowMapSMRTRayCount.Override(Labels[stage].Contains("8")?8:4);
  settings.virtualShadowMapSMRTSamplesPerRay.Override(8);
  settings.virtualShadowMapSMRTMaxRayLength.Override(10);
  File.WriteAllText(root+"/"+Labels[stage]+"/smrt-settings.json",JsonUtility.ToJson(settings,true));
  started=EditorApplication.timeSinceStartup;
 }
 static void Tick(){
  if(session==null)return;
  if(session.Phase=="Sampling" && lastFrame!=Time.frameCount && recorder.Valid && recorder.LastValue>0){
   lastFrame=Time.frameCount;scopedSamples.Add(recorder.LastValue*1e-6);
  }
  if(session.Tick()){
   File.WriteAllText(root+"/"+Labels[stage]+"/scoped-main-resolve-ms.json",JsonUtility.ToJson(new SampleReport{valid=recorder.Valid,values=scopedSamples.ToArray()}));
   session.Finish("completed");session.Dispose();session=null;
   if(++stage<Labels.Length){Begin();return;}
   File.WriteAllText(root+"/status.txt","complete");Finish();return;
  }
  if(session.IsFinished||EditorApplication.timeSinceStartup-started>60){File.WriteAllText(root+"/status.txt","incomplete stage="+stage);Finish();}
 }
 [Serializable] class SampleReport{public bool valid;public double[] values;}
 static void Finish(){
  EditorApplication.update-=Tick;AssemblyReloadEvents.beforeAssemblyReload-=Finish;
  if(session!=null){session.Dispose();session=null;}VirtualShadowMapPrototypeRuntime.EditorPhysicalPageLimit=originalLimit;
  if(light!=null)light.angularDiameter=originalAngle;
  recorder.Dispose();CSMShadowResolvePass.EditorBNDTimingCamera=0;CSMShadowResolvePass.EditorBNDLegacyPhase=0;CSMShadowResolvePass.EditorBNDSamplingFrame=-1;
  if(EditorApplication.isPlaying)EditorApplication.isPlaying=false;
 }
}}
