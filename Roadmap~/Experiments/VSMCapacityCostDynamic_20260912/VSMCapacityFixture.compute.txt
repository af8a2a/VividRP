#pragma kernel VSMCapacityFixture
#pragma kernel VSMCapacityClearLayers
#pragma use_dxc

RWTexture2DArray<uint> _FixtureStatic;
RWTexture2DArray<uint> _FixtureDynamic;
RWStructuredBuffer<uint> _FixtureOwners;
RWStructuredBuffer<uint> _FixtureTable;
RWStructuredBuffer<uint4> _FixtureMetadata;
uint _FixtureWidth;
uint _FixtureHeight;

[numthreads(8, 8, 1)]
void VSMCapacityClearLayers(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= _FixtureWidth || id.y >= _FixtureHeight || id.z >= 16) return;
    _FixtureStatic[id] = 0;
    _FixtureDynamic[id] = 0;
}

// Six physical slots, 4x4 texels/page, three slots/atlas row, two clipmaps,
// four virtual pages/clipmap. Deliberately include three owner failure classes.
uint VSMCapacityFixtureDepth(uint page, uint texel, uint layer, uint pool)
{
    if (page == 2) return layer == 0 ? asuint(0.75) : 0; // Unassigned stale data: excluded.
    if (page >= 3) return asuint(0.5); // Invalid owners: excluded from occupancy.
    if (page == 1)
        return pool == 0 ? (texel == 0 && layer == 0 ? asuint(0.75) : 0)
            : asuint(0.95 - 0.02 * layer);
    uint pattern = (texel + 7 * pool) % 16;
    if (pattern == 1) return layer < 3 ? asuint(0.9 - 0.1 * layer) : 0;
    if (pattern == 2) return asuint(0.95 - 0.02 * layer);
    if (pattern == 3) return layer < 2 ? asuint(0.8) : 0; // Adjacent duplicate.
    if (pattern == 4) return layer < 2 ? asuint(0.6 + 0.1 * layer) : 0; // Inversion.
    if (pattern == 5) return layer == 0 ? asuint(0.8) : (layer == 2 ? asuint(0.6) : 0); // Hole.
    if (pattern == 6) return layer == 0 ? asuint(1.5) : 0; // Invalid float depth bits.
    return 0;
}

[numthreads(64, 1, 1)]
void VSMCapacityFixture(uint3 id : SV_DispatchThreadID)
{
    uint i = id.x;
    if (i < 6)
        _FixtureOwners[i] = i == 0 ? 1 : i == 1 ? 5 : i == 2 ? 0 : i == 3 ? 999 : i == 4 ? 2 : 3;
    if (i < 8)
    {
        _FixtureTable[i] = i == 0 ? 1 : i == 1 ? 99 : i == 2 ? 6 : i == 4 ? 2 : 0;
        _FixtureMetadata[i] = uint4(i == 0 ? 6 : i == 2 ? 0 : 2,
            i == 0 ? 1 : i == 1 ? 99 : i == 2 ? 6 : i == 4 ? 2 : 0, 0, 0);
    }
    if (i >= 96) return;
    uint page = i / 16, texel = i % 16;
    uint2 p = uint2(page % 3, page / 3) * 4 + uint2(texel % 4, texel / 4);
    for (uint layer = 0; layer < 16; layer++)
    {
        _FixtureStatic[uint3(p, layer)] = VSMCapacityFixtureDepth(page, texel, layer, 0);
        _FixtureDynamic[uint3(p, layer)] = VSMCapacityFixtureDepth(page, texel, layer, 1);
    }
}
