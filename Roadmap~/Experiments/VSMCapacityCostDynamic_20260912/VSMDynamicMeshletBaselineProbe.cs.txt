        [MenuItem("Tools/VividRP/Diagnostics/Run VSM Dynamic Baseline")]
        private static void Run()
        {
            if (active != null) throw new InvalidOperationException("VSM dynamic capture is already active.");
            if (EditorApplication.isPlayingOrWillChangePlaymode) throw new InvalidOperationException("Use the idle Edit Mode scene.");
            Camera selected = null;
            foreach (var c in Object.FindObjectsByType<Camera>(FindObjectsSortMode.None))
                if (c.isActiveAndEnabled && c.cameraType == CameraType.Game && c.gameObject.scene.IsValid())
                { if (selected != null) throw new InvalidOperationException("Multiple active Game cameras."); selected = c; }
            if (selected == null) throw new InvalidOperationException("No active Game camera.");
            if (selected.GetComponent<VividAdditionalCameraData>() == null) throw new InvalidOperationException("Missing Vivid camera component.");
            Light sun = null;
            foreach (var l in Object.FindObjectsByType<Light>(FindObjectsSortMode.None))
                if (l.isActiveAndEnabled && l.type == LightType.Directional)
                { if (sun != null) throw new InvalidOperationException("Multiple directional lights."); sun = l; }
            if (sun == null) throw new InvalidOperationException("No directional light.");
            var config = File.Exists(RequestPath) ? JsonUtility.FromJson<Request>(File.ReadAllText(RequestPath)) : new Request();
            if (config.timingOnly)
            {
                // Fixed experiment: all scenarios forward and the same reverse.
                var forward = new Request().scenarios; config.scenarios = new string[forward.Length * 2];
                for (int i = 0; i < forward.Length; i++)
                { config.scenarios[i] = forward[i]; config.scenarios[forward.Length + i] = forward[forward.Length - 1 - i]; }
                config.modes = new[] { "adaptive" }; config.frames = 32; config.capacityReference = false;
            }
            if (config.frames < 2 || config.frames > 256 || config.referenceRays < 16 || config.referenceRays > 1024
                || config.stride < 1 || config.width < 1 || config.height < 1 || config.scenarios.Length == 0 || config.modes.Length == 0)
                throw new InvalidOperationException("Invalid baseline request.");
            active = new VSMDynamicBaselineProbe(selected, sun, config);
            try { active.Begin(); } catch { active.Dispose(); throw; }
        }

        [MenuItem("Tools/VividRP/Diagnostics/Cancel VSM Dynamic Baseline")]
        private static void Cancel() { if (active != null) { active.error = "cancelled"; active.Finish(); } }

        private VSMDynamicBaselineProbe(Camera c, Light l, Request config)
        {
            request = config; camera = c; light = l; additional = c.GetComponent<VividAdditionalCameraData>();
            cameraPosition = c.transform.position; cameraRotation = c.transform.rotation; lightRotation = l.transform.rotation;
            background = Application.runInBackground;
            root = Path.GetFullPath("Packages/VividRP/Temp~/vsm-baseline/" + (config.timingOnly ? "dynamic-timing/" : "dynamic/") + DateTime.UtcNow.ToString("yyyyMMdd_HHmmss_fff"));
        }

        private void Begin()
        {
            if (camera.cullingMask != -1 || light.cullingMask != -1)
                throw new InvalidOperationException("This baseline requires camera and light GameObject layer masks = Everything.");
            Directory.CreateDirectory(root);
            File.WriteAllText(root + "/request.json", JsonUtility.ToJson(request, true));
            File.WriteAllText(root + "/original-camera.json", EditorJsonUtility.ToJson(camera));
            File.WriteAllText(root + "/original-light.json", EditorJsonUtility.ToJson(light));
            File.WriteAllText(root + "/original-settings.json", JsonUtility.ToJson(VividVolumeManagerUtility.GetCascadedShadowSettingsVolume(), true));
            if (request.timingOnly) timing = new VSMStageTimingCapture(128);
            else
            {
                reference = Object.Instantiate(AssetDatabase.LoadAssetAtPath<ComputeShader>(AssetRoot + "VSMBaselineReference.compute"));
                if (reference == null) throw new InvalidOperationException("Reference shader is not imported.");
            }
            volumeObject = new GameObject("Temporary VSM Dynamic Baseline") { hideFlags = HideFlags.HideAndDontSave };
            var volume = volumeObject.AddComponent<Volume>(); volume.isGlobal = true; volume.priority = float.MaxValue;
            material = Object.Instantiate(Resources.Load<Material>("DefaultMaterial"));
            if (material == null) throw new InvalidOperationException("Default Vivid material unavailable.");
            material.hideFlags = HideFlags.HideAndDontSave; material.SetFloat("_Cull", 0);
            material.SetFloat("_ReceiveShadows", 1); material.SetFloat("_TransmissionWeight", 0);
            material.SetFloat("_ThinWalledTransmission", 0); material.SetFloat("_Surface", 0); material.SetFloat("_AlphaClip", 0);
            material.SetFloat("_Metallic", 0); material.SetFloat("_Smoothness", .5f); material.SetColor("_BaseColor", new Color(.5f, .5f, .5f, 1));
            material.DisableKeyword("_RECEIVE_SHADOWS_OFF"); material.DisableKeyword("_SURFACE_TYPE_TRANSPARENT");
            material.DisableKeyword("_ALPHATEST_ON"); material.SetFloat("_ZWrite", 1); material.renderQueue = 2000;
            gridWidth = (request.width + request.stride - 1) / request.stride;
            gridHeight = (request.height + request.stride - 1) / request.stride;
            if (!request.timingOnly) { visibility = MakeOutput(); signal = MakeOutput(); world = MakeOutput(); }
            // Fixed opaque fixture on Sponza's y=0 courtyard floor. Its world
            // position and actual receiver world coordinates are saved for QA.
            fixtureCenter = cameraPosition + cameraRotation * Vector3.forward * request.fixtureForwardDistance;
            fixtureCenter.y = request.floorHeight + .01f;
            Application.runInBackground = true;
            RenderPipelineManager.beginCameraRendering += BeforeCamera;
            CSMShadowResolvePass.EditorReceiverCapture += Capture;
            EditorApplication.update += Tick;
            AssemblyReloadEvents.beforeAssemblyReload += Dispose;
            EditorApplication.quitting += Dispose;
            StartStage();
        }

        private RenderTexture MakeOutput()
        {
            var rt = new RenderTexture(new RenderTextureDescriptor(gridWidth, gridHeight, GraphicsFormat.R32G32B32A32_SFloat, 0) { enableRandomWrite = true });
            rt.hideFlags = HideFlags.HideAndDontSave; rt.Create(); return rt;
        }

        private void StartStage()
        {
            capturing = stageFinished = geometryBuilt = false; warm = step = 0;
            timingPrepared = false; timing?.Reset();
            camera.transform.SetPositionAndRotation(cameraPosition, cameraRotation); light.transform.rotation = lightRotation;
            DestroyFixture();
            acceleration?.Dispose(); acceleration = null;
            fixtureReferenceHandle = -1;
            Object.DestroyImmediate(profile); profile = ScriptableObject.CreateInstance<VolumeProfile>(); profile.hideFlags = HideFlags.HideAndDontSave;
            var settings = profile.Add<CascadedShadowSettingsVolume>();
            settings.enableCSM.Override(true); settings.enableVirtualShadowMapPrototype.Override(true);
            settings.virtualShadowMapSMRT.Override(true);
            settings.screenSpaceShadowDenoise.Override(true); settings.virtualShadowMapSMRTTemporalDenoise.Override(true);
            settings.virtualShadowMapSMRTAdaptiveRays.Override(request.modes[modeIndex] == "adaptive");
            settings.virtualShadowMapSMRTRayCount.Override(request.modes[modeIndex] == "fixed8" ? 8 : 4);
            volumeObject.GetComponent<Volume>().sharedProfile = profile; additional.ResetPostProcessingHistory();
            directory = root + "/" + request.scenarios[scenarioIndex] + "_" + request.modes[modeIndex];
            if (request.timingOnly)
            {
                directory += scenarioIndex < request.scenarios.Length / 2 ? "_forward" : "_reverse"; Directory.CreateDirectory(directory);
                timingWindow = new TimingWindow { scenario = request.scenarios[scenarioIndex], window = scenarioIndex,
                    order = scenarioIndex < request.scenarios.Length / 2 ? "forward" : "reverse",
                    fixtureBackend = request.scenarios[scenarioIndex] == "deform" ? "meshlet prebaked frame sequence; registered renderer switch" : "meshlet rigid",
                    trajectory = "32-frame original-speed cycle during warm and observation; includes 31-to-0 reset, no slow motion",
                    gpuFrameIdentity = "ProfilerRecorder last completed profiling frame; observed camera IDs are timestamps, not GPU IDs; no quality-frame phase pairing" };
            }
            else
            {
                Directory.CreateDirectory(directory);
                framePrefixes = new string[request.frames];
                for (int i = 0; i < framePrefixes.Length; i++) framePrefixes[i] = directory + "/frame_" + i.ToString("D3");
            }
            BuildFixture(request.scenarios[scenarioIndex]); ApplyPose(0);
            if (!request.timingOnly) BuildGeometry();
            stageStarted = EditorApplication.timeSinceStartup;
            File.WriteAllText(root + "/progress.txt", Path.GetFileName(directory) + " warmup");
        }

        private void BuildFixture(string scenario)
        {
            if (scenario != "thin_sweep" && scenario != "deform" && scenario != "receiver_slide" && scenario != "receiver_lift") return;
            fixtureProxy = ScriptableObject.CreateInstance<GPUDrivenMaterialProxy>(); fixtureProxy.hideFlags = HideFlags.HideAndDontSave;
            fixtureProxy.name = "Temporary VSM Opaque Gray Proxy"; fixtureProxy.SourceMaterial = material;
            fixtureProxy.Model = GPUDrivenMaterialProxyModel.StandardLit; fixtureProxy.BaseColor = new Color(.5f, .5f, .5f, 1);
            fixtureProxy.Metallic = 0; fixtureProxy.Roughness = .5f; fixtureProxy.MaskMode = GPUDrivenMaterialMaskMode.None;
            fixtureProxy.CullMode = CullMode.Off; fixtureProxy.AlphaClip = false; fixtureProxy.DisableLighting = false;
            if (scenario == "deform")
            {
                fixture = new GameObject("Temporary VSM Deforming Sheet");
                fixture.hideFlags = HideFlags.HideInHierarchy;
                bakedMeshes = new Mesh[request.frames]; bakedVertices = new Vector3[request.frames][];
                bakedRenderers = new MeshletRenderer[request.frames]; bakedSourceRenderers = new MeshRenderer[request.frames];
                fixtureCollections = new VividMeshletCollectionAsset[request.frames];
                var uv = new Vector2[21 * 9];
                var indices = new int[20 * 8 * 6]; int next = 0;
                for (int y = 0; y < 9; y++) for (int x = 0; x < 21; x++)
                { int v = y * 21 + x; uv[v] = new Vector2(x / 20f, y / 8f); if (x == 20 || y == 8) continue;
                  indices[next++] = v; indices[next++] = v + 21; indices[next++] = v + 1;
                  indices[next++] = v + 1; indices[next++] = v + 21; indices[next++] = v + 22; }
                for (int frame = 0; frame < request.frames; frame++)
                {
                    var child = new GameObject("Temporary VSM Baked Sheet " + frame); child.hideFlags = HideFlags.HideInHierarchy;
                    child.SetActive(false); child.transform.SetParent(fixture.transform, false);
                    var vertices = new Vector3[21 * 9]; float phase = Mathf.Clamp01((frame - 4f) / Math.Max(1f, request.frames - 9f)) * 6.283185f;
                    for (int y = 0; y < 9; y++) for (int x = 0; x < 21; x++)
                    { float u = x / 20f, v = y / 8f; vertices[y * 21 + x] = new Vector3((u - .5f) * 2f, v * 1.8f, .3f * Mathf.Sin(phase + u * 6.283185f) * v); }
                    var mesh = new Mesh { name = "Temporary VSM Baked Mesh " + frame, hideFlags = HideFlags.HideAndDontSave };
                    mesh.vertices = vertices; mesh.uv = uv; mesh.triangles = indices; mesh.RecalculateNormals(); mesh.RecalculateTangents(); mesh.RecalculateBounds();
                    bakedMeshes[frame] = mesh; bakedVertices[frame] = vertices;
                    child.AddComponent<MeshFilter>().sharedMesh = mesh;
                    var source = child.AddComponent<MeshRenderer>(); ConfigureSource(source, true);
                    bakedSourceRenderers[frame] = source;
                    bakedRenderers[frame] = BindMeshlet(child, source, mesh, frame);
                }
                SelectBakedFrame(0);
            }
            else
            {
                fixture = GameObject.CreatePrimitive(PrimitiveType.Cube); Object.DestroyImmediate(fixture.GetComponent<Collider>());
                fixture.SetActive(false); fixture.hideFlags = HideFlags.HideInHierarchy;
                fixture.name = "Temporary VSM " + scenario;
                fixture.transform.localScale = scenario == "thin_sweep" ? new Vector3(.045f, 2f, .06f) : new Vector3(2.5f, scenario == "receiver_lift" ? .04f : .004f, 2f);
                fixtureRenderer = fixture.GetComponent<MeshRenderer>();
                ConfigureSource(fixtureRenderer, false); fixtureCollections = new VividMeshletCollectionAsset[1];
                fixtureMeshlet = BindMeshlet(fixture, fixtureRenderer, fixture.GetComponent<MeshFilter>().sharedMesh, 0);
                fixture.SetActive(true);
            }
        }

        private void ConfigureSource(MeshRenderer source, bool deformed)
        {
            source.sharedMaterial = material; source.shadowCastingMode = ShadowCastingMode.TwoSided; source.receiveShadows = true;
            source.rayTracingMode = deformed ? RayTracingMode.DynamicGeometryManualUpdate : RayTracingMode.DynamicTransform;
        }

        private MeshletRenderer BindMeshlet(GameObject go, MeshRenderer source, Mesh mesh, int index)
        {
            var collection = ScriptableObject.CreateInstance<VividMeshletCollectionAsset>(); collection.hideFlags = HideFlags.HideAndDontSave;
            collection.name = "Temporary VSM Meshlets " + index; fixtureCollections[index] = collection;
            var parameters = new VividMeshletCollectionBuilder.Parameters { Mesh = mesh, SubMeshIndex = 0,
                MaxMeshLODLevelCount = 1, OptimizeVertexCache = false, PreserveFinestLODLevels = true };
            VividMeshletCollectionBuilder.Generate(collection, in parameters);
            var renderer = go.AddComponent<MeshletRenderer>(); renderer.CaptureSourceFromRenderer(source);
            renderer.SetMeshletCollections(new[] { collection }); renderer.SetMaterialProxies(new[] { fixtureProxy });
            if (!renderer.TryValidate(out string message)) throw new InvalidOperationException("Fixture meshlet bindings invalid: " + message);
            source.enabled = false; // Only the meshlet backend writes visible/depth/shadow surfaces.
            return renderer;
        }

        private void SelectBakedFrame(int frame)
        {
            if (bakedIndex == frame) return;
            if (bakedIndex >= 0) bakedRenderers[bakedIndex].gameObject.SetActive(false);
            bakedIndex = frame; fixtureMeshlet = bakedRenderers[frame]; fixtureRenderer = bakedSourceRenderers[frame];
            deformMesh = bakedMeshes[frame]; deformVertices = bakedVertices[frame];
            fixtureMeshlet.gameObject.SetActive(true);
        }

        private void DestroyFixture()
        {
            Object.DestroyImmediate(fixture); fixture = null; fixtureRenderer = null; fixtureMeshlet = null;
            if (fixtureCollections != null) foreach (var collection in fixtureCollections) Object.DestroyImmediate(collection);
            if (bakedMeshes != null) foreach (var mesh in bakedMeshes) Object.DestroyImmediate(mesh);
            Object.DestroyImmediate(fixtureProxy); fixtureProxy = null; fixtureCollections = null;
            bakedMeshes = null; bakedVertices = null; bakedRenderers = null; bakedSourceRenderers = null;
            bakedIndex = -1; deformMesh = null; deformVertices = null;
        }

        private void ApplyPose(int frame)
        {
            float motion = Mathf.Clamp01((frame - 4f) / Math.Max(1f, request.frames - 9f));
            string scenario = request.scenarios[scenarioIndex];
            camera.transform.SetPositionAndRotation(cameraPosition, cameraRotation); light.transform.rotation = lightRotation;
            if (scenario == "camera_slide") camera.transform.position = cameraPosition + cameraRotation * Vector3.right * (motion * .7f);
            if (scenario == "camera_turn") camera.transform.rotation = Quaternion.AngleAxis(motion * 7f, Vector3.up) * cameraRotation;
            if (scenario == "sun_rotate") light.transform.rotation = Quaternion.AngleAxis(motion * 3f, Vector3.up) * lightRotation;
            if (scenario == "sun_step") light.transform.rotation = Quaternion.AngleAxis(frame >= 8 ? 3f : 0f, Vector3.up) * lightRotation;
            if (fixture != null)
            {
                fixture.transform.position = fixtureCenter;
                if (scenario == "thin_sweep") fixture.transform.position += new Vector3(-1.5f + 3f * motion, 1f, 0);
                if (scenario == "receiver_slide") fixture.transform.position = new Vector3(fixtureCenter.x - .7f + 1.4f * motion, request.floorHeight + .004f, fixtureCenter.z);
                if (scenario == "receiver_lift") fixture.transform.position = new Vector3(fixtureCenter.x - .7f + 1.4f * motion, request.floorHeight + .05f + .6f * motion, fixtureCenter.z);
                if (scenario == "deform") SelectBakedFrame(frame);
                VividMeshletRendererDatabase.instance.UpdateRendererTransformData(fixtureMeshlet);
            }
        }

        private void BuildGeometry()
        {
            acceleration = new RayTracingAccelerationStructure(new RayTracingAccelerationStructure.Settings(
                RayTracingAccelerationStructure.ManagementMode.Manual, RayTracingAccelerationStructure.RayTracingModeMask.Everything, -1));
            int staticInstances = 0, dynamicInstances = 0, triangles = 0;
            uint shadowLayers = (uint)light.GetComponent<VividAdditionalLightData>().effectiveShadowRenderingLayers;
            foreach (var r in Object.FindObjectsByType<MeshletRenderer>(FindObjectsSortMode.None))
            {
                if (!r.isActiveAndEnabled || !r.sourceRenderingEnabled || r.shadowCastingMode == ShadowCastingMode.Off || r.sourceMesh == null) continue;
                if (r == fixtureMeshlet) continue; // Added separately with an updateable mesh handle below.
                if ((r.renderingLayerMask & shadowLayers) == 0) throw new InvalidOperationException("Reference caster outside light rendering layers.");
                uint mask = r.gameObject.isStatic && !r.sourceWasSkinned ? 1u : 2u;
                for (int sub = 0; sub < r.sourceMesh.subMeshCount; sub++)
                {
                    var cfg = new RayTracingMeshInstanceConfig { mesh = r.sourceMesh, meshLod = -1, subMeshIndex = (uint)sub,
                        material = r.sourceMaterials[sub], mask = mask, enableTriangleCulling = false,
                        subMeshFlags = RayTracingSubMeshFlags.Enabled | RayTracingSubMeshFlags.ClosestHitOnly };
                    acceleration.AddInstance(in cfg, r.transform.localToWorldMatrix, null, 0);
                    triangles += (int)r.sourceMesh.GetIndexCount(sub) / 3;
                    if (mask == 1) staticInstances++; else dynamicInstances++;
                }
            }
            foreach (var r in Object.FindObjectsByType<MeshRenderer>(FindObjectsSortMode.None))
            {
                if (!r.enabled || !r.gameObject.activeInHierarchy || r.shadowCastingMode == ShadowCastingMode.Off) continue;
                var meshlet = r.GetComponent<MeshletRenderer>(); if (meshlet != null && meshlet.isActiveAndEnabled && meshlet.takeOverSourceRenderer) continue;
                var mf = r.GetComponent<MeshFilter>(); if (mf == null || mf.sharedMesh == null) continue;
                if (((uint)r.renderingLayerMask & shadowLayers) == 0) throw new InvalidOperationException("Reference Renderer outside light rendering layers.");
                var flags = new RayTracingSubMeshFlags[mf.sharedMesh.subMeshCount];
                for (int sub = 0; sub < flags.Length; sub++) flags[sub] = RayTracingSubMeshFlags.Enabled | RayTracingSubMeshFlags.ClosestHitOnly;
                int handle = acceleration.AddInstance(r, flags, false, false, 2);
                if (handle < 0) throw new InvalidOperationException("Reference Renderer AddInstance failed.");
                dynamicInstances++;
                for (int sub = 0; sub < mf.sharedMesh.subMeshCount; sub++) triangles += (int)mf.sharedMesh.GetIndexCount(sub) / 3;
            }
            if (fixtureMeshlet != null)
            { UpdateFixtureReference(); dynamicInstances++; triangles += (int)fixtureMeshlet.sourceMesh.GetIndexCount(0) / 3; }
            File.WriteAllText(directory + "/geometry.txt", "staticInstances=" + staticInstances + " dynamicInstances=" + dynamicInstances + " triangles=" + triangles
                + "\nmask1=static meshlet && !skinned; mask2=dynamic meshlet and all Unity Renderer\nforce opaque; no terrain/procedural/animated skinned reference; fixture=" + request.scenarios[scenarioIndex]
                + "\nfixtureBackend=" + (bakedMeshes != null ? "meshlet prebaked frame sequence; registered renderer switch" : "meshlet rigid")
                + "\nfixtureReference=explicit MeshInstanceConfig mask2 Enabled|ClosestHitOnly; remove/reAdd at every capture + UpdateInstanceGeometry(handle)");
        }

        private void UpdateFixtureReference()
        {
            if (fixtureReferenceHandle >= 0) acceleration.RemoveInstance(fixtureReferenceHandle);
            var cfg = new RayTracingMeshInstanceConfig { mesh = fixtureMeshlet.sourceMesh, meshLod = -1, subMeshIndex = 0,
                material = material, mask = 2, enableTriangleCulling = false,
                subMeshFlags = RayTracingSubMeshFlags.Enabled | RayTracingSubMeshFlags.ClosestHitOnly,
                rayTracingMode = RayTracingMode.DynamicGeometryManualUpdate };
            fixtureReferenceHandle = acceleration.AddInstance(in cfg, fixtureMeshlet.transform.localToWorldMatrix, null, 0);
            if (fixtureReferenceHandle < 0) throw new InvalidOperationException("Reference fixture MeshInstanceConfig AddInstance failed.");
            acceleration.UpdateInstanceGeometry(fixtureReferenceHandle);
        }

        private void BeforeCamera(ScriptableRenderContext context, Camera current)
        {
            if (disposed || stageFinished || current != camera) return;
            ApplyPose(request.timingOnly ? warm & 31 : capturing ? step : bakedMeshes != null && warm < 64 ? warm % request.frames : 0);
        }

        private void Tick()
        {
            if (disposed) return;
            try
            {
            if (camera == null || light == null || EditorApplication.isPlayingOrWillChangePlaymode) error = "Scene or editor mode changed.";
            if (error != null || EditorApplication.timeSinceStartup - stageStarted > 300) { error ??= "stage timeout"; Finish(); return; }
            if (stageFinished && pending == 0)
            {
                if (request.timingOnly)
                {
                    timing.WriteCsv(directory + "/stage-timing.csv");
                    File.WriteAllText(directory + "/window.json", JsonUtility.ToJson(timingWindow, true));
                    File.WriteAllText(directory + "/settings-final.json", JsonUtility.ToJson(VividVolumeManagerUtility.GetCascadedShadowSettingsVolume(), true));
                }
                if (++modeIndex == request.modes.Length) { modeIndex = 0; scenarioIndex++; }
                if (scenarioIndex == request.scenarios.Length) { Finish(); return; }
                StartStage();
            }
            EditorApplication.QueuePlayerLoopUpdate(); UnityEditorInternal.InternalEditorUtility.RepaintAllViews();
            }
            catch (Exception exception) { error = exception.ToString(); Finish(); }
        }

        private void Capture(ComputePassContext context, Texture depth, Texture normal, Texture shadow)
        {
            try { CaptureFrame(context, depth, normal, shadow); }
            catch (Exception e) { error = e.ToString(); stageFinished = true; }
        }

        private void CaptureFrame(ComputePassContext context, Texture depth, Texture normal, Texture shadow)
        {
            var c = context.Get<VividCameraData>();
            if (disposed || stageFinished || c.camera != camera || c.frameIndex == lastFrame) return;
            lastFrame = c.frameIndex;
            if (request.timingOnly) { CaptureTimingFrame(context, c, depth); return; }
            if (++warm <= 64) return;
            if (!VirtualShadowMapPrototypeRuntime.IsFrameActive) { error = "VSM inactive: " + VirtualShadowMapPrototypeRuntime.LastFallbackReason; return; }
            if (VirtualShadowMapPrototypeRuntime.PhysicalPageCapacity != 256 || VirtualShadowMapPrototypeRuntime.VirtualResolution != 2048
                || VirtualShadowMapPrototypeRuntime.DepthLayerCount != 16) { error = "Unexpected VSM layout; expected 2048 / 256 / 16."; return; }
            if (!capturing)
            {
                if ((c.frameIndex & 255) != 0) return;
                capturing = true; firstFrame = c.frameIndex;
                File.WriteAllText(root + "/progress.txt", Path.GetFileName(directory) + " capturing");
                File.WriteAllText(directory + "/settings.json", JsonUtility.ToJson(VividVolumeManagerUtility.GetCascadedShadowSettingsVolume(), true));
            }
            if (c.frameIndex - firstFrame != step) { error = "Non-consecutive receiver frames."; return; }
            if (request.originX + (gridWidth - 1) * request.stride >= depth.width || request.originY + (gridHeight - 1) * request.stride >= depth.height)
            { error = "Reference ROI exceeds actual receiver size."; return; }
            var cmd = context.cmd.m_WrappedCommandBuffer;
            if (fixtureRenderer != null)
            {
                UpdateFixtureReference();
            }
            if (!geometryBuilt || fixtureRenderer != null) { cmd.BuildRayTracingAccelerationStructure(acceleration); geometryBuilt = true; }
            var lightData = context.Get<VividLightData>(); var direction = -lightData.mainLight.transform.forward;
            var settings = VividVolumeManagerUtility.GetCascadedShadowSettingsVolume();
            int expectedRays = request.modes[modeIndex] == "fixed8" ? 8 : 4;
            if (VirtualShadowMapReceiverQuality.BuildSMRTParameters(settings, light.GetComponent<VividAdditionalLightData>().angularDiameter).x != expectedRays
                || !settings.screenSpaceShadowDenoise.value || !settings.virtualShadowMapSMRTTemporalDenoise.value
                || settings.virtualShadowMapSMRTAdaptiveRays.value != (request.modes[modeIndex] == "adaptive"))
                throw new InvalidOperationException("Actual SMRT/temporal/adaptive configuration differs from the requested case.");
            int kernel = reference.FindKernel("ReferenceReceivers");
            DirectionalRayTracedShadowPass.ComputeSunBasis(direction, out var basisX, out var basisY);
            cmd.SetRayTracingAccelerationStructure(reference, kernel, "_ReferenceAcceleration", acceleration);
            cmd.SetComputeTextureParam(reference, kernel, "_ReferenceDepthTexture", depth);
            cmd.SetComputeTextureParam(reference, kernel, "_ReferenceGBuffer1", normal);
            cmd.SetComputeTextureParam(reference, kernel, "_ReferenceVisibility", visibility);
            cmd.SetComputeTextureParam(reference, kernel, "_BaselineShadowTexture", shadow);
            cmd.SetComputeTextureParam(reference, kernel, "_BaselineSignal", signal);
            cmd.SetComputeTextureParam(reference, kernel, "_BaselineWorld", world);
            var history = camera.GetVividCameraHistory();
            bool hasHistory = history.TryGetTexture(CameraHistoryId.Create("VSMShadowHistory"), out var historyTexture) && historyTexture.IsValid();
            if (!hasHistory) throw new InvalidOperationException("No written temporal history after warmup.");
            cmd.SetComputeTextureParam(reference, kernel, "_BaselineHistory", hasHistory ? historyTexture.GetFrame(0).rt : normal);
            cmd.SetComputeIntParam(reference, "_BaselineHasHistory", hasHistory ? 1 : 0);
            cmd.SetComputeMatrixParam(reference, "_ReferenceInvViewProjection", c.GetGPUViewProjectionMatrix(true).inverse);
            cmd.SetComputeVectorParam(reference, "_ReferenceLightDirection", direction);
            cmd.SetComputeVectorParam(reference, "_ReferenceSunBasisX", basisX); cmd.SetComputeVectorParam(reference, "_ReferenceSunBasisY", basisY);
            cmd.SetComputeVectorParam(reference, "_ReferenceNormalBias", new Vector4(.001f, .01f, 0, 0));
            cmd.SetComputeFloatParam(reference, "_ReferenceTanAngularRadius", Mathf.Tan(light.GetComponent<VividAdditionalLightData>().angularDiameter * Mathf.Deg2Rad * .5f));
            cmd.SetComputeFloatParam(reference, "_ReferenceRayMaxDistance", 1000f);
            cmd.SetComputeIntParam(reference, "_ReferenceScreenWidth", depth.width); cmd.SetComputeIntParam(reference, "_ReferenceScreenHeight", depth.height);
            cmd.SetComputeIntParam(reference, "_ReferenceROIOriginX", request.originX); cmd.SetComputeIntParam(reference, "_ReferenceROIOriginY", request.originY);
            cmd.SetComputeIntParam(reference, "_ReferenceGridWidth", gridWidth); cmd.SetComputeIntParam(reference, "_ReferenceGridHeight", gridHeight);
            cmd.SetComputeIntParam(reference, "_ReferencePixelStride", request.stride); cmd.SetComputeIntParam(reference, "_ReferenceRayCount", request.referenceRays);
            cmd.DispatchCompute(reference, kernel, (gridWidth + 7) / 8, (gridHeight + 7) / 8, 1);
            string prefix = framePrefixes[step];
            Save(cmd, visibility, prefix + "_reference.gz"); Save(cmd, signal, prefix + "_signal.gz"); Save(cmd, world, prefix + "_world.gz");
            Save(cmd, VirtualShadowMapPrototypeRuntime.AllocatorCounters, prefix + "_pages.gz");
            var frame = new Frame { step = step, frame = c.frameIndex, phase = c.frameIndex & 255,
                width = depth.width, height = depth.height, gridWidth = gridWidth, gridHeight = gridHeight,
                pages = VirtualShadowMapPrototypeRuntime.PhysicalPageCapacity, levels = VirtualShadowMapPrototypeRuntime.Projections.Count,
                referenceRays = request.referenceRays, scenario = request.scenarios[scenarioIndex], mode = request.modes[modeIndex],
                state = VirtualShadowMapPrototypeRuntime.FrameState.ToString(), fallback = VirtualShadowMapPrototypeRuntime.LastFallbackReason.ToString(),
                cameraPosition = camera.transform.position, cameraEuler = camera.transform.eulerAngles, lightDirection = direction,
                fixturePosition = fixture != null ? fixture.transform.position : Vector3.zero, viewProjection = c.GetGPUViewProjectionMatrix(true),
                quality = VirtualShadowMapReceiverQuality.BuildParameters(settings),
                smrt = VirtualShadowMapReceiverQuality.BuildSMRTParameters(settings, light.GetComponent<VividAdditionalLightData>().angularDiameter), historyPresent = hasHistory,
                fixturePhase = Mathf.Clamp01((step - 4f) / Math.Max(1f, request.frames - 9f)) * 6.283185f,
                fixtureGeometryHash = GeometryHash(), fixtureBackend = bakedMeshes != null ? "meshlet prebaked frame sequence" : fixture != null ? "meshlet rigid" : "none" };
            File.WriteAllText(prefix + ".json", JsonUtility.ToJson(frame));
            if (step == 0) additional.RequestFinalFrameScreenshot(directory + "/first.png");
            if (step == request.frames - 1) additional.RequestFinalFrameScreenshot(directory + "/last.png");
            if (step == 0 && request.capacityReference) CaptureCapacity(cmd, prefix);
            step++; stageFinished = step >= request.frames;
        }

        private void CaptureTimingFrame(ComputePassContext context, VividCameraData c, Texture depth)
        {
            if (!VirtualShadowMapPrototypeRuntime.IsFrameActive
                || VirtualShadowMapPrototypeRuntime.PhysicalPageCapacity != 256
                || VirtualShadowMapPrototypeRuntime.VirtualResolution != 2048 || VirtualShadowMapPrototypeRuntime.DepthLayerCount != 16)
                throw new InvalidOperationException("Timing requires active VSM with 2048 / 256 / 16 layout.");
            var settings = VividVolumeManagerUtility.GetCascadedShadowSettingsVolume();
            var sun = light.GetComponent<VividAdditionalLightData>();
            var smrt = VirtualShadowMapReceiverQuality.BuildSMRTParameters(settings, sun.angularDiameter);
            if (!settings.virtualShadowMapSMRT.value || smrt.x != 4 || !settings.screenSpaceShadowDenoise.value
                || !settings.virtualShadowMapSMRTTemporalDenoise.value || !settings.virtualShadowMapSMRTAdaptiveRays.value)
                throw new InvalidOperationException("Timing requires adaptive4 with spatial and short-history denoise.");
            if (!timingPrepared)
            {
                // One setup callback writes effective metadata before the 64 warm
                // callbacks. It does not consume a trajectory or observation step.
                timingWindow.fixtureRayTracingMode = fixtureMeshlet != null ? "production dynamic meshlet; no diagnostic RTAS" : "none";
                timingWindow.fixtureShader = fixtureRenderer != null ? material.shader.name : "none";
                timingWindow.fixtureTwoSided = fixtureRenderer != null && fixtureRenderer.shadowCastingMode == ShadowCastingMode.TwoSided;
                timingWindow.fixtureReceivesShadows = fixtureRenderer != null && fixtureRenderer.receiveShadows;
                timingWindow.fixtureSceneCaster = fixture != null && fixture.scene.IsValid() && (fixture.hideFlags & HideFlags.DontSave) == 0;
                timingWindow.fixtureRenderingLayers = fixtureRenderer != null ? fixtureRenderer.renderingLayerMask : 0;
                timingWindow.lightShadowLayers = (uint)sun.effectiveShadowRenderingLayers;
                if (fixtureRenderer != null && (!timingWindow.fixtureSceneCaster || fixtureMeshlet == null || !fixtureMeshlet.isActiveAndEnabled || fixtureRenderer.enabled
                    || (timingWindow.fixtureRenderingLayers & timingWindow.lightShadowLayers) == 0
                    || !fixtureMeshlet.TryValidate(out _)))
                    throw new InvalidOperationException("Timing fixture is not an eligible meshlet caster/receiver.");
                UpdateTimingMetadata(context, c, depth, settings, smrt);
                File.WriteAllText(directory + "/warm-start.json", JsonUtility.ToJson(timingWindow, true));
                File.WriteAllText(directory + "/settings.json", JsonUtility.ToJson(settings, true));
                timingPrepared = true; return;
            }
            if (depth.width != timingWindow.width || depth.height != timingWindow.height
                || smrt != timingWindow.smrt || VirtualShadowMapReceiverQuality.BuildParameters(settings) != timingWindow.quality)
                throw new InvalidOperationException("Timing receiver size or effective quality changed during a window.");
            if (warm == 0) firstFrame = timingWindow.firstWarmFrame = c.frameIndex;
            if (c.frameIndex - firstFrame != warm) throw new InvalidOperationException("Non-consecutive timing camera frames.");
            warm++;
            timing.Observe(c.frameIndex, EditorApplication.timeSinceStartup, true);
            if (warm <= 64)
            {
                timingWindow.lastWarmFrame = c.frameIndex;
                if (warm == 64) timing.Reset();
                return;
            }
            if (step == 0) timingWindow.firstObservedFrame = c.frameIndex;
            step = timing.Count; timingWindow.observed = step; timingWindow.lastObservedFrame = c.frameIndex;
            if (step == 128)
            {
                UpdateTimingMetadata(context, c, depth, settings, smrt);
                if (!timingWindow.historyPresent) throw new InvalidOperationException("Timing window has no valid temporal history.");
                stageFinished = true;
            }
        }

        private void UpdateTimingMetadata(ComputePassContext context, VividCameraData c, Texture depth, CascadedShadowSettingsVolume settings, Vector4 smrt)
        {
            timingWindow.metadataFrame = c.frameIndex; timingWindow.width = depth.width; timingWindow.height = depth.height;
            timingWindow.resolution = VirtualShadowMapPrototypeRuntime.VirtualResolution;
            timingWindow.capacity = VirtualShadowMapPrototypeRuntime.PhysicalPageCapacity;
            timingWindow.depthLayers = VirtualShadowMapPrototypeRuntime.DepthLayerCount;
            timingWindow.clipmaps = VirtualShadowMapPrototypeRuntime.Projections.Count;
            timingWindow.cameraPosition = camera.transform.position; timingWindow.cameraEuler = camera.transform.eulerAngles;
            timingWindow.lightDirection = -light.transform.forward;
            timingWindow.fixturePosition = fixture != null ? fixture.transform.position : Vector3.zero;
            timingWindow.fixtureScale = fixture != null ? fixture.transform.localScale : Vector3.zero;
            var sd = context.Get<VividShadowData>();
            timingWindow.receiver = new Vector4(settings.virtualShadowMapPCF.value ? 1 : 0, sd.depthBias, sd.slopeScaleDepthBias,
                settings.virtualShadowMapStochasticFiltering.value ? 1 : 0);
            timingWindow.quality = VirtualShadowMapReceiverQuality.BuildParameters(settings); timingWindow.smrt = smrt;
            timingWindow.sampleIndexOffset = VirtualShadowMapReceiverQuality.BuildSMRTSampleIndexOffset(settings, additional, c.frameIndex);
            timingWindow.aa = context.Get<VividAntialiasingData>().effectiveMode.ToString();
            timingWindow.historyPresent = camera.GetVividCameraHistory().TryGetTexture(CameraHistoryId.Create("VSMShadowHistory"), out var history) && history.IsValid();
        }

        private void CaptureCapacity(CommandBuffer cmd, string prefix)
        {
            int capacity = VirtualShadowMapPrototypeRuntime.PhysicalPageCapacity;
            int count = capacity * request.capacityGrid * request.capacityGrid * 2;
            if (capacityStats == null)
            {
                capacityStats = new ComputeBuffer(count, 16);
                capacityErrors = new ComputeBuffer(count, 16);
                capacityDepths = new ComputeBuffer(count * 64, 4);
            }
            int kernel = reference.FindKernel("ReferenceCapacity");
            cmd.SetRayTracingAccelerationStructure(reference, kernel, "_ReferenceAcceleration", acceleration);
            cmd.SetComputeTextureParam(reference, kernel, "_VSMPrototypeStaticPhysicalPage", VirtualShadowMapPrototypeRuntime.StaticPhysicalPage);
            cmd.SetComputeTextureParam(reference, kernel, "_VSMPrototypeDynamicPhysicalPage", VirtualShadowMapPrototypeRuntime.DynamicPhysicalPage);
            cmd.SetComputeBufferParam(reference, kernel, "_VSMPrototypePhysicalPageOwners", VirtualShadowMapPrototypeRuntime.PhysicalPageOwners);
            cmd.SetComputeBufferParam(reference, kernel, "_VSMPrototypePageTable", VirtualShadowMapPrototypeRuntime.PageTable);
            cmd.SetComputeBufferParam(reference, kernel, "_VSMProjections", VirtualShadowMapPrototypeRuntime.Projections.Buffer);
            cmd.SetComputeBufferParam(reference, kernel, "_CapacityCounts", capacityStats);
            cmd.SetComputeBufferParam(reference, kernel, "_CapacityDiagnostics", capacityErrors);
            cmd.SetComputeBufferParam(reference, kernel, "_CapacityReferenceDepths", capacityDepths);
            cmd.SetComputeIntParam(reference, "_VSMPrototypePageSize", VirtualShadowMapPrototypeRuntime.PageSize);
            cmd.SetComputeIntParam(reference, "_VSMPrototypePagesPerAxis", VirtualShadowMapPrototypeRuntime.PagesPerAxis);
            cmd.SetComputeIntParam(reference, "_VSMPrototypeVirtualResolution", VirtualShadowMapPrototypeRuntime.VirtualResolution);
            cmd.SetComputeIntParam(reference, "_VSMPrototypePhysicalPagesPerRow", VirtualShadowMapPrototypeRuntime.PhysicalPagesPerRow);
            cmd.SetComputeIntParam(reference, "_VSMPrototypePhysicalPageCapacity", capacity);
            cmd.SetComputeIntParam(reference, "_VSMPrototypePageTableEntryCount", VirtualShadowMapPrototypeRuntime.PageTable.count);
            cmd.SetComputeIntParam(reference, "_VSMProjectionCount", VirtualShadowMapPrototypeRuntime.Projections.Count);
            cmd.SetComputeIntParam(reference, "_CapacitySamplesPerPageAxis", request.capacityGrid);
            cmd.SetComputeIntParam(reference, "_CapacityStaticMask", 1); cmd.SetComputeIntParam(reference, "_CapacityDynamicMask", 2);
            cmd.SetComputeFloatParam(reference, "_CapacityUniqueDepthToleranceWorld", .0001f);
            cmd.SetComputeFloatParam(reference, "_CapacityProductionMatchToleranceWorld", .01f);
            cmd.DispatchCompute(reference, kernel, (request.capacityGrid + 7) / 8, (request.capacityGrid + 7) / 8, capacity);
            Save(cmd, capacityStats, prefix + "_capacity-counts.gz"); Save(cmd, capacityErrors, prefix + "_capacity-errors.gz");
            Save(cmd, capacityDepths, prefix + "_capacity-reference-depths.gz");
            Save(cmd, VirtualShadowMapPrototypeRuntime.PhysicalPageOwners, prefix + "_owners.gz");
            Save(cmd, VirtualShadowMapPrototypeRuntime.Projections.Buffer, prefix + "_projections.gz");
            Save(cmd, VirtualShadowMapPrototypeRuntime.PageTable, prefix + "_table.gz");
        }

        private uint GeometryHash()
        {
            if (deformMesh == null) return 0;
            uint hash = 2166136261;
            unchecked
            {
                for (int i = 0; i < deformVertices.Length; i++)
                {
                    hash = (hash ^ (uint)BitConverter.SingleToInt32Bits(deformVertices[i].x)) * 16777619;
                    hash = (hash ^ (uint)BitConverter.SingleToInt32Bits(deformVertices[i].y)) * 16777619;
                    hash = (hash ^ (uint)BitConverter.SingleToInt32Bits(deformVertices[i].z)) * 16777619;
                }
            }
            return hash;
        }

        private void Save(CommandBuffer cmd, Texture texture, string path)
        { pending++; cmd.RequestAsyncReadback(texture, 0, TextureFormat.RGBAFloat, request => Complete(request, path)); }
        private void Save(CommandBuffer cmd, GraphicsBuffer buffer, string path)
        { pending++; cmd.RequestAsyncReadback(buffer, request => Complete(request, path)); }
        private void Save(CommandBuffer cmd, ComputeBuffer buffer, string path)
        { pending++; cmd.RequestAsyncReadback(buffer, request => Complete(request, path)); }
        private void Complete(AsyncGPUReadbackRequest readback, string path)
        {
            try { if (readback.hasError) error = "GPU readback failed: " + path;
                else { using var file = File.Create(path); using var zip = new GZipStream(file, System.IO.Compression.CompressionLevel.Fastest);
                    var bytes = readback.GetData<byte>().ToArray(); zip.Write(bytes, 0, bytes.Length); } }
            catch (Exception e) { error = e.ToString(); } finally { pending--; }
        }

        private void Finish() { try { File.WriteAllText(root + "/status.txt", error ?? "complete"); } finally { Dispose(); } }
        public void Dispose()
        {
            if (disposed) return; disposed = true;
            RenderPipelineManager.beginCameraRendering -= BeforeCamera; CSMShadowResolvePass.EditorReceiverCapture -= Capture;
            EditorApplication.update -= Tick; AssemblyReloadEvents.beforeAssemblyReload -= Dispose;
            EditorApplication.quitting -= Dispose;
            if (!request.timingOnly) AsyncGPUReadback.WaitAllRequests();
            timing?.Dispose();
            if (camera != null) camera.transform.SetPositionAndRotation(cameraPosition, cameraRotation);
            if (light != null) light.transform.rotation = lightRotation;
            acceleration?.Dispose(); capacityStats?.Dispose(); capacityErrors?.Dispose(); capacityDepths?.Dispose();
            DestroyFixture(); Object.DestroyImmediate(material);
            Object.DestroyImmediate(volumeObject); Object.DestroyImmediate(profile); Object.DestroyImmediate(reference);
            Object.DestroyImmediate(visibility); Object.DestroyImmediate(signal); Object.DestroyImmediate(world);
            Application.runInBackground = background; additional?.ResetPostProcessingHistory(); active = null;
            EditorApplication.QueuePlayerLoopUpdate();
        }
    }
