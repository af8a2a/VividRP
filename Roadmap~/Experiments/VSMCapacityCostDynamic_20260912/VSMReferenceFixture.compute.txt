#pragma kernel UploadReferencePools
#pragma use_dxc

// Fixture upload preserves uint depth bits exactly and uses the same LoadStore
// resource capability as production VSM arrays. No format conversion is used.
StructuredBuffer<uint> _ReferenceFixtureDepthBits;
RWTexture2DArray<uint> _ReferenceFixtureStatic;
RWTexture2DArray<uint> _ReferenceFixtureDynamic;

[numthreads(32, 1, 1)]
void UploadReferencePools(uint3 id : SV_DispatchThreadID)
{
    uint index = id.x;
    if (index >= 32u)
        return;
    uint3 texel = uint3(0u, 0u, index % 16u);
    if (index < 16u)
        _ReferenceFixtureStatic[texel] = _ReferenceFixtureDepthBits[index];
    else
        _ReferenceFixtureDynamic[texel] = _ReferenceFixtureDepthBits[index];
}
