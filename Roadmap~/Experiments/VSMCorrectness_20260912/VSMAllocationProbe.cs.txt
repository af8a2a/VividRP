using System;
using System.IO;
using UnityEditor;
using VividRP.Runtime.RenderPass.Core;
namespace VividRP.Editor {
internal static class VSMAllocationProbe {
 static int frames; static long total,maximum; static bool active; static double started;
 [MenuItem("Tools/VividRP/Diagnostics/Measure Temporary VSM Allocations")]
 static void Run() {if(active)throw new Exception("Already active");frames=0;total=maximum=0;active=true;started=EditorApplication.timeSinceStartup;CSMShadowResolvePass.EditorResolveAllocationCapture+=Measure;EditorApplication.update+=Tick;AssemblyReloadEvents.beforeAssemblyReload+=Stop;}
 static void Measure(long bytes) { if(frames++<64)return;total+=bytes;maximum=Math.Max(maximum,bytes); }
 static void Tick() {if(frames>=192){File.WriteAllText("Packages/VividRP/Temp~/vsm-correctness/allocations.txt","warmup=64 samples="+(frames-64)+" bytes="+total+" max="+maximum);Stop();if(total!=0)throw new Exception("VSM resolve allocated");return;}if(EditorApplication.timeSinceStartup-started>120){File.WriteAllText("Packages/VividRP/Temp~/vsm-correctness/allocations.txt","timeout");Stop();return;}EditorApplication.QueuePlayerLoopUpdate();UnityEditorInternal.InternalEditorUtility.RepaintAllViews();}
 static void Stop(){CSMShadowResolvePass.EditorResolveAllocationCapture-=Measure;EditorApplication.update-=Tick;AssemblyReloadEvents.beforeAssemblyReload-=Stop;active=false;}
}}
