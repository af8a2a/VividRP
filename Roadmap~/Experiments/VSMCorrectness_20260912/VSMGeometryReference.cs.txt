using System;
using System.IO;
using System.IO.Compression;
using UnityEditor;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Experimental.Rendering;
using VividRP.Runtime;
using VividRP.Runtime.GPUDriven;
using VividRP.Runtime.RenderPass.Core;
using Object=UnityEngine.Object;
namespace VividRP.Editor {
internal sealed class VSMGeometryReference : IDisposable {
 static VSMGeometryReference active;
 Camera camera; Vector3 position; Quaternion rotation; bool background,disposed; int step,pending,last=-1; double started;
 RayTracingAccelerationStructure acceleration;
 ComputeShader shader; RenderTexture output; string root,error;
 [MenuItem("Tools/VividRP/Diagnostics/Capture Temporary VSM Geometry")]
 static void Run() { if(active!=null)throw new Exception("Active"); active=new VSMGeometryReference(); try{active.Begin();}catch{active.Dispose();throw;} }
 void Begin() {
  foreach(var c in Object.FindObjectsByType<Camera>(FindObjectsSortMode.None)) if(c.isActiveAndEnabled&&c.cameraType==CameraType.Game&&c.gameObject.scene.IsValid()){if(camera!=null)throw new Exception("Multiple cameras");camera=c;}
  if(camera==null)throw new Exception("No camera");
  position=camera.transform.position; rotation=camera.transform.rotation; background=Application.runInBackground; Application.runInBackground=true;
  root=Path.GetFullPath("Packages/VividRP/Temp~/vsm-correctness/reference/"+DateTime.UtcNow.ToString("yyyyMMdd_HHmmss_fff"));Directory.CreateDirectory(root);
  shader=Object.Instantiate(AssetDatabase.LoadAssetAtPath<ComputeShader>("Packages/com.vivid.render-pipelines/Editor/Tools/VSMGeometryReference.compute"));
  acceleration=new RayTracingAccelerationStructure(new RayTracingAccelerationStructure.Settings(RayTracingAccelerationStructure.ManagementMode.Manual,RayTracingAccelerationStructure.RayTracingModeMask.Everything,-1));
  int meshes=0,triangles=0;
  foreach(var r in Object.FindObjectsByType<MeshletRenderer>(FindObjectsSortMode.None)) {
   if(!r.isActiveAndEnabled||!r.sourceRenderingEnabled||r.shadowCastingMode==ShadowCastingMode.Off||r.sourceMesh==null)continue;
   for(int sub=0;sub<r.sourceMesh.subMeshCount;sub++) {
    var cfg=new RayTracingMeshInstanceConfig{mesh=r.sourceMesh,meshLod=-1,subMeshIndex=(uint)sub,material=r.sourceMaterials[sub],mask=255,enableTriangleCulling=false,subMeshFlags=RayTracingSubMeshFlags.Enabled|RayTracingSubMeshFlags.ClosestHitOnly};
    acceleration.AddInstance(in cfg,r.transform.localToWorldMatrix,null,0);triangles+=(int)r.sourceMesh.GetIndexCount(sub)/3;
   } meshes++;
  }
  foreach(var r in Object.FindObjectsByType<MeshRenderer>(FindObjectsSortMode.None)) {
   if(!r.enabled||!r.gameObject.activeInHierarchy||r.shadowCastingMode==ShadowCastingMode.Off)continue;
   var meshlet=r.GetComponent<MeshletRenderer>();if(meshlet!=null&&meshlet.isActiveAndEnabled&&meshlet.takeOverSourceRenderer)continue;
   var mf=r.GetComponent<MeshFilter>();if(mf==null||mf.sharedMesh==null)continue;
   acceleration.AddInstance(r,null,false,false,255);meshes++;triangles+=(int)mf.sharedMesh.GetIndexCount(0)/3;
  }
  File.WriteAllText(root+"/geometry.txt","meshes="+meshes+" triangles="+triangles+" instances="+acceleration.GetInstanceCount()+" opaque triangle reference; alpha silhouettes excluded from metrics");
  started=EditorApplication.timeSinceStartup; CSMShadowResolvePass.EditorReceiverCapture+=Capture; EditorApplication.update+=Tick; AssemblyReloadEvents.beforeAssemblyReload+=Dispose;
 }
 void Tick() {
  if(disposed)return;
  if(camera.transform.position!=position||Quaternion.Angle(camera.transform.rotation,rotation)>.0001)error="Camera moved";
  if(error!=null||EditorApplication.timeSinceStartup-started>120){File.WriteAllText(root+"/status.txt",error??"timeout");Dispose();return;}
  if(step==32&&pending==0){File.WriteAllText(root+"/status.txt","complete");Dispose();return;}
  EditorApplication.QueuePlayerLoopUpdate();UnityEditorInternal.InternalEditorUtility.RepaintAllViews();
 }
 void Capture(ComputePassContext ctx,Texture depth,Texture normal,Texture shadow) {
  var c=ctx.Get<VividCameraData>(); if(c.camera!=camera||step>=32||c.frameIndex==last||acceleration==null)return;last=c.frameIndex;
  var cmd=ctx.cmd.m_WrappedCommandBuffer;if(step==0)cmd.BuildRayTracingAccelerationStructure(acceleration); var ld=ctx.Get<VividLightData>(); var dir=-ld.mainLight.transform.forward;
  if(output==null){output=new RenderTexture(new RenderTextureDescriptor(depth.width/4,depth.height/4,GraphicsFormat.R32G32B32A32_SFloat,0){enableRandomWrite=true});output.Create();
   var stats=VividRayTracingAccelerationStructureStatsRegistry.LastStats;
   File.WriteAllText(root+"/state.txt","camera="+camera.name+" position="+position.ToString("R")+" rotation="+rotation.ToString("R")+" size="+depth.width+"x"+depth.height+" instances="+acceleration.GetInstanceCount()+" culling="+stats.CullingMode+" candidates="+stats.CandidateRendererCount+" status="+stats.StatusMessage);
  }
  int k=shader.FindKernel("Reference");DirectionalRayTracedShadowPass.ComputeSunBasis(dir,out var bx,out var by);
  cmd.SetRayTracingAccelerationStructure(shader,k,"_AccelerationStructure",acceleration);
  cmd.SetComputeTextureParam(shader,k,"_DepthTexture",depth);cmd.SetComputeTextureParam(shader,k,"_GBuffer1",normal);cmd.SetComputeTextureParam(shader,k,"_Output",output);
  cmd.SetComputeMatrixParam(shader,"_InvViewProjectionMatrix",c.GetGPUViewProjectionMatrix(true).inverse);
  cmd.SetComputeVectorParam(shader,"_LightDirectionWS",dir);cmd.SetComputeVectorParam(shader,"_SunBasisX",bx);cmd.SetComputeVectorParam(shader,"_SunBasisY",by);
  cmd.SetComputeFloatParam(shader,"_TanSunAngularRadius",Mathf.Tan(ld.mainLight.GetComponent<VividAdditionalLightData>().angularDiameter*Mathf.Deg2Rad*.5f));
  cmd.SetComputeIntParam(shader,"_OutputWidth",depth.width);cmd.SetComputeIntParam(shader,"_OutputHeight",depth.height);cmd.SetComputeIntParam(shader,"_Batch",step);
  cmd.DispatchCompute(shader,k,(output.width+7)/8,(output.height+7)/8,1);
  string file=root+"/frame_"+step.ToString("D3")+".bin.gz";pending++;cmd.RequestAsyncReadback(output,0,TextureFormat.RGBAFloat,r=>{try{if(r.hasError)error="Readback failed";else{using var f=File.Create(file);using var z=new GZipStream(f,System.IO.Compression.CompressionLevel.Fastest);byte[] b=r.GetData<byte>().ToArray();z.Write(b,0,b.Length);}}catch(Exception e){error=e.ToString();}finally{pending--;}});step++;
 }
 public void Dispose(){if(disposed)return;disposed=true;CSMShadowResolvePass.EditorReceiverCapture-=Capture;EditorApplication.update-=Tick;AssemblyReloadEvents.beforeAssemblyReload-=Dispose;AsyncGPUReadback.WaitAllRequests();acceleration?.Dispose();Object.DestroyImmediate(output);Object.DestroyImmediate(shader);Application.runInBackground=background;active=null;EditorApplication.QueuePlayerLoopUpdate();}
}}
