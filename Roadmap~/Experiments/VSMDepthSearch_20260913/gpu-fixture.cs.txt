// Direct GPU validation through Unity CLI; does not invoke Unity Test Framework.
var source = UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.ComputeShader>("Packages/com.vivid.render-pipelines/Tests/Editor/RenderPass/Shadows/VirtualShadowMapSamplingTests.compute");
var shader = UnityEngine.Object.Instantiate(source);
UnityEngine.RenderTexture MakePool() {
    var d = new UnityEngine.RenderTextureDescriptor(16,16,UnityEngine.Experimental.Rendering.GraphicsFormat.R32_UInt,0) {
        dimension=UnityEngine.Rendering.TextureDimension.Tex2DArray,volumeDepth=16,enableRandomWrite=true,msaaSamples=1 };
    var t = new UnityEngine.RenderTexture(d); if(!t.Create()) throw new System.Exception("Pool creation failed"); return t;
}
var stat = MakePool(); var dyn = MakePool();
var staticData = new uint[4096]; var dynamicData = new uint[4096];
const int count = 32768;
var inputs = new UnityEngine.Vector4[count]; var rays = new UnityEngine.Vector4[count]; var results = new UnityEngine.Vector2[count];
var sb = new UnityEngine.GraphicsBuffer(UnityEngine.GraphicsBuffer.Target.Structured,4096,4);
var db = new UnityEngine.GraphicsBuffer(UnityEngine.GraphicsBuffer.Target.Structured,4096,4);
var ib = new UnityEngine.GraphicsBuffer(UnityEngine.GraphicsBuffer.Target.Structured,count,16);
var rb = new UnityEngine.GraphicsBuffer(UnityEngine.GraphicsBuffer.Target.Structured,count,16);
var ob = new UnityEngine.GraphicsBuffer(UnityEngine.GraphicsBuffer.Target.Structured,count,8);
int mismatches = 0, hits = 0, checkedCount = 0;
try {
    int upload = shader.FindKernel("UploadTestPools"), inspect = shader.FindKernel("InspectSMRTDepthIntervals");
    shader.GetKernelThreadGroupSizes(upload,out uint ux,out _,out _);
    shader.GetKernelThreadGroupSizes(inspect,out uint ix,out _,out _);
    if (ux != 8 || ix != 64) throw new System.Exception("Invalid diagnostic kernels");
    shader.SetBuffer(upload,"_TestStaticData",sb); shader.SetBuffer(upload,"_TestDynamicData",db);
    shader.SetTexture(upload,"_TestStaticPool",stat); shader.SetTexture(upload,"_TestDynamicPool",dyn);
    shader.SetBuffer(inspect,"_SamplingInputs",ib); shader.SetBuffer(inspect,"_SamplingNormals",rb); shader.SetBuffer(inspect,"_SamplingResults",ob);
    shader.SetTexture(inspect,"_VSMPrototypeStaticPhysicalPage",stat); shader.SetTexture(inspect,"_VSMPrototypeDynamicPhysicalPage",dyn);
    shader.SetInt("_SamplingCount",count);
    for (int batch = 0; batch < 4; batch++) {
        System.Array.Clear(staticData,0,staticData.Length); System.Array.Clear(dynamicData,0,dynamicData.Length);
        for (int pixel = 0; pixel < 256; pixel++) {
            int pair = (pixel + batch * 256) % 289, sc = pair % 17, dc = pair / 17;
            for (int layer = 0; layer < 16; layer++) {
                if (layer < sc) staticData[layer*256+pixel] = Unity.Mathematics.math.asuint(.99f-.041f*layer);
                if (layer < dc) dynamicData[layer*256+pixel] = Unity.Mathematics.math.asuint(.973f-.043f*layer);
            }
            for (int c = 0; c < 128; c++) {
                int i = pixel*128+c;
                float origin = batch%2==0?.07f:.49f, scale = batch<2?.05f:.00003125f;
                uint raw = ((c&1)==0?staticData:dynamicData)[c/8*256+pixel];
                float center = (Unity.Mathematics.math.asfloat(raw)-origin)/scale;
                float enter = UnityEngine.Mathf.Max(0,center+(c%4-2)*.00001f);
                inputs[i] = new UnityEngine.Vector4(pixel%16,pixel/16,enter,enter+(c%7)*.1f);
                rays[i] = new UnityEngine.Vector4(origin,scale,(c%5)*.01f,0);
            }
        }
        sb.SetData(staticData); db.SetData(dynamicData); shader.Dispatch(upload,2,2,16);
        ib.SetData(inputs); rb.SetData(rays); shader.Dispatch(inspect,count/64,1,1); ob.GetData(results);
        for (int i=0;i<count;i++) { if(results[i].x!=results[i].y) mismatches++; if(results[i].x>0) hits++; checkedCount++; }
    }
    var report = new {checkedCount,mismatches,hits,misses=checkedCount-hits,depthLayers=16,prefixPairs=289};
    System.IO.File.WriteAllText("E:/VividRP_Reborn/Packages/VividRP/Roadmap~/Experiments/VSMDepthSearch_20260913/gpu-fixture.json",Newtonsoft.Json.JsonConvert.SerializeObject(report));
    if(mismatches!=0 || hits==0 || hits==checkedCount) throw new System.Exception("Interval validation failed");
    return report;
} finally {
    sb.Dispose();db.Dispose();ib.Dispose();rb.Dispose();ob.Dispose();stat.Release();dyn.Release();
    UnityEngine.Object.DestroyImmediate(stat);UnityEngine.Object.DestroyImmediate(dyn);UnityEngine.Object.DestroyImmediate(shader);
}
