using System;
using System.IO;
using UnityEditor;
using UnityEngine;
using UnityEngine.Experimental.Rendering;
using UnityEngine.Rendering;
using VividRP.Runtime;
using VividRP.Runtime.RenderPass.Core;
using Object = UnityEngine.Object;

namespace VividRP.Editor
{
    // Explicit live diagnostic. No Unity tests, scene saves, Play Mode switches,
    // camera movement, or geometry changes. Keep capacity and timing runs separate.
    internal sealed class VSMBaselineCostProbe : IDisposable
    {
        private static VSMBaselineCostProbe s_Active;
        private static readonly string[] s_TimingCases = {
            "baseline_forward", "skip_forward", "skip_reverse", "baseline_reverse" };
        private static readonly int[] s_TimingModes = { 2, 2, 2, 2 };
        private const int WarmFrames = 64, TimingFrames = 128;
        private readonly Camera m_Camera;
        private readonly VividAdditionalCameraData m_Additional;
        private readonly bool m_TimingOnly, m_Background;
        private readonly Vector3 m_Position;
        private readonly Quaternion m_Rotation;
        private readonly float m_Fov, m_OrthoSize, m_Near, m_Far;
        private readonly bool m_Orthographic;
        private readonly string m_Root;
        private GameObject m_VolumeObject;
        private VolumeProfile m_Profile;
        private ComputeShader m_Audit;
        private GraphicsBuffer m_Pages, m_Summary;
        private VSMStageTimingCapture m_Timing;
        private Light m_Light;
        private Quaternion m_LightRotation;
        private float m_LightAngle;
        private int m_PagesKernel, m_ReduceKernel, m_Stage, m_Warm, m_Step, m_Pending, m_LastFrame = -1;
        private int m_Width, m_Height;
        private double m_Started;
        private string m_Directory, m_Error;
        private bool m_Disposed;

        [Serializable] private sealed class Frame
        {
            public int frame, width, height, resolution, capacity, pageSize, pagesPerAxis,
                physicalPagesPerRow, projectionCount, tableEntries, projectionBufferCount, projectionStride,
                layerCount, staticWidth, staticHeight, dynamicWidth, dynamicHeight;
            public long staticDepthBytes, dynamicDepthBytes, rasterDepthBytes, unityRasterDepthBytes;
            public string state, fallback, aa;
            public Matrix4x4 viewProjection, inverseViewProjection;
            public Vector3 cameraPosition, cameraEuler, lightDirection;
            public Vector4 receiver, quality, smrt;
            public int sampleIndexOffset;
        }

        [MenuItem("Tools/VividRP/Diagnostics/Run Temporary VSM Baseline Capacity")]
        private static void RunCapacity() => Run(false);
        [MenuItem("Tools/VividRP/Diagnostics/Run Temporary VSM Baseline Timing")]
        private static void RunTiming() => Run(true);
        [MenuItem("Tools/VividRP/Diagnostics/Cancel Temporary VSM Baseline Cost")]
        private static void Cancel() { if (s_Active != null) s_Active.Finish("cancelled"); }

        private static void Run(bool timing)
        {
            if (s_Active != null) throw new InvalidOperationException("VSM baseline cost capture already active.");
            Camera selected = null;
            foreach (Camera candidate in Object.FindObjectsByType<Camera>(FindObjectsSortMode.None))
                if (candidate.cameraType == CameraType.Game && candidate.isActiveAndEnabled && candidate.gameObject.scene.IsValid())
                {
                    if (selected != null) throw new InvalidOperationException("Multiple active Game cameras; capture requires a unique camera.");
                    selected = candidate;
                }
            if (selected == null) throw new InvalidOperationException("No active Game camera.");
            s_Active = new VSMBaselineCostProbe(selected, timing);
            try { s_Active.Begin(); }
            catch { s_Active.Dispose(); throw; }
        }

        private VSMBaselineCostProbe(Camera camera, bool timing)
        {
            m_Camera = camera; m_Additional = camera.GetComponent<VividAdditionalCameraData>();
            m_TimingOnly = timing; m_Background = Application.runInBackground;
            m_Position = camera.transform.position; m_Rotation = camera.transform.rotation;
            m_Fov = camera.fieldOfView; m_OrthoSize = camera.orthographicSize;
            m_Near = camera.nearClipPlane; m_Far = camera.farClipPlane; m_Orthographic = camera.orthographic;
            m_Root = Path.GetFullPath("Packages/VividRP/Temp~/vsm-empty-skip/cost/" + DateTime.UtcNow.ToString("yyyyMMdd_HHmmss_fff"));
        }

        private void Begin()
        {
            Directory.CreateDirectory(m_Root);
            File.WriteAllText(Path.GetDirectoryName(m_Root) + "/latest.txt", m_Root);
            File.WriteAllText(m_Root + "/mode.txt", m_TimingOnly ? "timing" : "capacity");
            File.WriteAllText(m_Root + "/camera.json", EditorJsonUtility.ToJson(m_Camera));
            if (m_Additional != null) File.WriteAllText(m_Root + "/additional.json", EditorJsonUtility.ToJson(m_Additional));
            File.WriteAllText(m_Root + "/original-settings.json", JsonUtility.ToJson(VividVolumeManagerUtility.GetCascadedShadowSettingsVolume()));
            if (m_TimingOnly) m_Timing = new VSMStageTimingCapture(TimingFrames);
            else
            {
                var source = AssetDatabase.LoadAssetAtPath<ComputeShader>("Packages/com.vivid.render-pipelines/Editor/Tools/VSMCapacityAudit.compute");
                if (source == null) throw new InvalidOperationException("VSMCapacityAudit.compute is not imported.");
                m_Audit = Object.Instantiate(source); m_Audit.hideFlags = HideFlags.HideAndDontSave;
                m_PagesKernel = m_Audit.FindKernel("VSMCapacityAuditPages");
                m_ReduceKernel = m_Audit.FindKernel("VSMCapacityReduce");
            }
            m_VolumeObject = new GameObject("Temporary VSM Baseline Cost") { hideFlags = HideFlags.HideAndDontSave };
            var volume = m_VolumeObject.AddComponent<Volume>(); volume.isGlobal = true; volume.priority = float.MaxValue;
            Application.runInBackground = true;
            CSMShadowResolvePass.EditorReceiverCapture += Capture;
            EditorApplication.update += Tick;
            AssemblyReloadEvents.beforeAssemblyReload += OnReload;
            EditorApplication.quitting += OnQuit;
            StartStage();
        }

        private void StartStage()
        {
            if (m_Profile != null) Object.DestroyImmediate(m_Profile);
            m_Profile = ScriptableObject.CreateInstance<VolumeProfile>(); m_Profile.hideFlags = HideFlags.HideAndDontSave;
            var settings = m_Profile.Add<CascadedShadowSettingsVolume>();
            settings.enableCSM.Override(true);
            settings.enableVirtualShadowMapPrototype.Override(true);
            settings.virtualShadowMapSMRT.Override(true);
            if (m_TimingOnly)
            {
                int mode = s_TimingModes[m_Stage];
                AssetDatabase.LoadAssetAtPath<ComputeShader>("Packages/com.vivid.render-pipelines/Shaders/Core/Private/CSMShadowResolve.compute").SetInt("_VSMPageOccupancySkipDisabled", m_Stage == 0 || m_Stage == 3 ? 1 : 0);
                settings.screenSpaceShadowDenoise.Override(true);
                settings.virtualShadowMapSMRTTemporalDenoise.Override(mode >= 1);
                settings.virtualShadowMapSMRTAdaptiveRays.Override(mode == 2);
            }
            m_VolumeObject.GetComponent<Volume>().sharedProfile = m_Profile;
            m_Directory = m_Root + "/" + (m_TimingOnly ? s_TimingCases[m_Stage] : "capacity");
            Directory.CreateDirectory(m_Directory);
            m_Warm = m_Step = 0; m_Started = EditorApplication.timeSinceStartup;
            m_Timing?.Reset();
            File.WriteAllText(m_Root + "/progress.txt", m_TimingOnly ? s_TimingCases[m_Stage] : "capacity");
        }

        private void Tick()
        {
            if (m_Disposed) return;
            try
            {
                if (m_Camera == null || m_Camera.transform.position != m_Position
                    || Quaternion.Angle(m_Camera.transform.rotation, m_Rotation) > .0001f
                    || m_Camera.fieldOfView != m_Fov || m_Camera.orthographicSize != m_OrthoSize
                    || m_Camera.orthographic != m_Orthographic || m_Camera.nearClipPlane != m_Near || m_Camera.farClipPlane != m_Far)
                    m_Error = "Camera pose or projection changed during the static baseline.";
                if (m_Light != null && Quaternion.Angle(m_Light.transform.rotation, m_LightRotation) > .0001f)
                    m_Error = "Light direction changed during the static baseline.";
                if (m_Error != null) { Finish("failed: " + m_Error); return; }
                if (EditorApplication.timeSinceStartup - m_Started > 120) { Finish("failed: stage timeout"); return; }
                if (m_Step >= (m_TimingOnly ? TimingFrames : 1) && m_Pending == 0)
                {
                    if (m_TimingOnly) m_Timing.WriteCsv(m_Directory + "/stage-timing.csv");
                    if (!m_TimingOnly || ++m_Stage == s_TimingCases.Length) { Finish("complete"); return; }
                    StartStage();
                }
                EditorApplication.QueuePlayerLoopUpdate();
                UnityEditorInternal.InternalEditorUtility.RepaintAllViews();
            }
            catch (Exception exception) { Finish("failed: " + exception); }
        }

        private void Capture(ComputePassContext context, Texture depth, Texture normal, Texture shadow)
        {
            if (m_Disposed || m_Error != null) return;
            var cameraData = context.Get<VividCameraData>();
            if (cameraData.camera != m_Camera || cameraData.frameIndex == m_LastFrame
                || m_Step >= (m_TimingOnly ? TimingFrames : 1)) return;
            m_LastFrame = cameraData.frameIndex;
            try
            {
                m_Warm++;
                if (m_Warm == 1)
                {
                    File.WriteAllText(m_Directory + "/settings.json", JsonUtility.ToJson(VividVolumeManagerUtility.GetCascadedShadowSettingsVolume()));
                    File.WriteAllText(m_Directory + "/warm-start-frame.json", JsonUtility.ToJson(BuildFrame(context, shadow)));
                }
                if (m_TimingOnly && m_Warm <= WarmFrames)
                {
                    m_Timing.Observe(cameraData.frameIndex, EditorApplication.timeSinceStartup, VirtualShadowMapPrototypeRuntime.IsFrameActive);
                    if (m_Warm == WarmFrames) m_Timing.Reset();
                }
                if (m_Warm <= WarmFrames) return;
                ValidateBaseline(context, shadow);
                if (m_TimingOnly)
                {
                    m_Timing.Observe(cameraData.frameIndex, EditorApplication.timeSinceStartup, true);
                    m_Step = m_Timing.Count;
                }
                else
                {
                    File.WriteAllText(m_Directory + "/frame.json", JsonUtility.ToJson(BuildFrame(context, shadow)));
                    CaptureCapacity(context.cmd.m_WrappedCommandBuffer);
                    m_Step = 1;
                }
            }
            catch (Exception exception) { m_Error = exception.ToString(); }
        }

        private void ValidateBaseline(ComputePassContext context, Texture shadow)
        {
            if (!VirtualShadowMapPrototypeRuntime.IsFrameActive)
                throw new InvalidOperationException("VSM not active: " + VirtualShadowMapPrototypeRuntime.LastFallbackReason);
            if (VirtualShadowMapPrototypeRuntime.VirtualResolution != 2048
                || VirtualShadowMapPrototypeRuntime.PhysicalPageCapacity != 256 || VirtualShadowMapPrototypeRuntime.DepthLayerCount != 16)
                throw new InvalidOperationException("Baseline must retain actual 2048 resolution / 256 pages / 16 layers.");
            if (m_Width != shadow.width || m_Height != shadow.height)
                throw new InvalidOperationException("Render dimensions changed after stage snapshot.");
            var light = context.Get<VividLightData>().mainLight;
            if (light != m_Light || light == null || light.GetComponent<VividAdditionalLightData>().angularDiameter != m_LightAngle)
                throw new InvalidOperationException("Directional light or angular diameter changed.");
        }

        private Frame BuildFrame(ComputePassContext context, Texture shadow)
        {
            var c = context.Get<VividCameraData>(); var sd = context.Get<VividShadowData>();
            var light = context.Get<VividLightData>().mainLight;
            var settings = VividVolumeManagerUtility.GetCascadedShadowSettingsVolume();
            float angle = light.GetComponent<VividAdditionalLightData>().angularDiameter;
            if (m_Light == null) { m_Light = light; m_LightRotation = light.transform.rotation; m_LightAngle = angle; }
            if (m_Width == 0) { m_Width = shadow.width; m_Height = shadow.height; }
            RenderTexture staticPool = VirtualShadowMapPrototypeRuntime.StaticPhysicalPage.rt;
            RenderTexture dynamicPool = VirtualShadowMapPrototypeRuntime.DynamicPhysicalPage.rt;
            RenderTexture raster = VirtualShadowMapPrototypeRuntime.RasterDepth.rt;
            RenderTexture unityRaster = VirtualShadowMapPrototypeRuntime.UnityRasterDepth.rt;
            Matrix4x4 vp = c.GetGPUViewProjectionMatrix(true);
            return new Frame {
                frame = c.frameIndex, width = shadow.width, height = shadow.height,
                resolution = VirtualShadowMapPrototypeRuntime.VirtualResolution, capacity = VirtualShadowMapPrototypeRuntime.PhysicalPageCapacity,
                pageSize = VirtualShadowMapPrototypeRuntime.PageSize, pagesPerAxis = VirtualShadowMapPrototypeRuntime.PagesPerAxis,
                physicalPagesPerRow = VirtualShadowMapPrototypeRuntime.PhysicalPagesPerRow,
                projectionCount = VirtualShadowMapPrototypeRuntime.Projections.Count, tableEntries = VirtualShadowMapPrototypeRuntime.PageTable.count,
                projectionBufferCount = VirtualShadowMapPrototypeRuntime.Projections.Buffer.count,
                projectionStride = VirtualShadowMapPrototypeRuntime.Projections.Buffer.stride,
                layerCount = staticPool.volumeDepth, staticWidth = staticPool.width, staticHeight = staticPool.height,
                dynamicWidth = dynamicPool.width, dynamicHeight = dynamicPool.height,
                staticDepthBytes = (long)staticPool.width * staticPool.height * staticPool.volumeDepth * 4,
                dynamicDepthBytes = (long)dynamicPool.width * dynamicPool.height * dynamicPool.volumeDepth * 4,
                rasterDepthBytes = (long)raster.width * raster.height * raster.volumeDepth * 4,
                unityRasterDepthBytes = (long)unityRaster.width * unityRaster.height * unityRaster.volumeDepth * 4,
                state = VirtualShadowMapPrototypeRuntime.FrameState.ToString(), fallback = VirtualShadowMapPrototypeRuntime.LastFallbackReason.ToString(),
                aa = context.Get<VividAntialiasingData>().effectiveMode.ToString(), viewProjection = vp, inverseViewProjection = vp.inverse,
                cameraPosition = m_Camera.transform.position, cameraEuler = m_Camera.transform.eulerAngles, lightDirection = -light.transform.forward,
                receiver = new Vector4(settings.virtualShadowMapPCF.value ? 1 : 0, sd.depthBias, sd.slopeScaleDepthBias, settings.virtualShadowMapStochasticFiltering.value ? 1 : 0),
                quality = VirtualShadowMapReceiverQuality.BuildParameters(settings), smrt = VirtualShadowMapReceiverQuality.BuildSMRTParameters(settings, angle),
                sampleIndexOffset = VirtualShadowMapReceiverQuality.BuildSMRTSampleIndexOffset(settings, m_Additional, c.frameIndex)
            };
        }

        private void CaptureCapacity(CommandBuffer cmd)
        {
            int capacity = VirtualShadowMapPrototypeRuntime.PhysicalPageCapacity;
            int levels = VirtualShadowMapPrototypeRuntime.Projections.Count;
            int size = VirtualShadowMapPrototypeRuntime.PageSize;
            int row = VirtualShadowMapPrototypeRuntime.PhysicalPagesPerRow;
            int entries = VirtualShadowMapPrototypeRuntime.PageTable.count;
            int axis = VirtualShadowMapPrototypeRuntime.PagesPerAxis;
            var staticPool = VirtualShadowMapPrototypeRuntime.StaticPhysicalPage.rt;
            var dynamicPool = VirtualShadowMapPrototypeRuntime.DynamicPhysicalPage.rt;
            if (capacity <= 0 || levels <= 0 || size <= 0 || row <= 0 || axis <= 0 || entries <= 0
                || entries != levels * axis * axis
                || VirtualShadowMapPrototypeRuntime.PhysicalPageOwners.count != capacity
                || VirtualShadowMapPrototypeRuntime.PageMetadata.count != entries)
                throw new InvalidOperationException("Invalid capacity audit dimensions.");
            ValidatePool(staticPool, capacity, row, size); ValidatePool(dynamicPool, capacity, row, size);
            m_Pages = new GraphicsBuffer(GraphicsBuffer.Target.Structured, 2 * capacity * 64, 4);
            m_Summary = new GraphicsBuffer(GraphicsBuffer.Target.Structured, 2 * (levels + 1) * 64, 4);
            cmd.SetComputeIntParam(m_Audit, "_VSMPrototypePageSize", size);
            cmd.SetComputeIntParam(m_Audit, "_VSMPrototypePagesPerAxis", axis);
            cmd.SetComputeIntParam(m_Audit, "_VSMPrototypePhysicalPagesPerRow", row);
            cmd.SetComputeIntParam(m_Audit, "_VSMPrototypePageTableEntryCount", entries);
            cmd.SetComputeIntParam(m_Audit, "_VSMPrototypePhysicalPageCapacity", capacity);
            cmd.SetComputeIntParam(m_Audit, "_VSMProjectionCount", levels);
            cmd.SetComputeTextureParam(m_Audit, m_PagesKernel, "_VSMPrototypeStaticPhysicalPage", staticPool);
            cmd.SetComputeTextureParam(m_Audit, m_PagesKernel, "_VSMPrototypeDynamicPhysicalPage", dynamicPool);
            cmd.SetComputeBufferParam(m_Audit, m_PagesKernel, "_VSMPrototypePhysicalPageOwners", VirtualShadowMapPrototypeRuntime.PhysicalPageOwners);
            cmd.SetComputeBufferParam(m_Audit, m_PagesKernel, "_VSMPrototypePageTable", VirtualShadowMapPrototypeRuntime.PageTable);
            cmd.SetComputeBufferParam(m_Audit, m_PagesKernel, "_VSMPrototypePageMetadata", VirtualShadowMapPrototypeRuntime.PageMetadata);
            cmd.SetComputeBufferParam(m_Audit, m_PagesKernel, "_VSMCapacityPages", m_Pages);
            cmd.DispatchCompute(m_Audit, m_PagesKernel, capacity, 2, 1);
            cmd.SetComputeBufferParam(m_Audit, m_ReduceKernel, "_VSMCapacityPages", m_Pages);
            cmd.SetComputeBufferParam(m_Audit, m_ReduceKernel, "_VSMCapacitySummary", m_Summary);
            cmd.DispatchCompute(m_Audit, m_ReduceKernel, levels + 1, 2, 1);
            Save(cmd, m_Pages, "pages.bin"); Save(cmd, m_Summary, "summary.bin");
            Save(cmd, VirtualShadowMapPrototypeRuntime.PhysicalPageOwners, "owners.bin");
            Save(cmd, VirtualShadowMapPrototypeRuntime.PageTable, "pagetable.bin");
            Save(cmd, VirtualShadowMapPrototypeRuntime.PageMetadata, "metadata.bin");
            Save(cmd, VirtualShadowMapPrototypeRuntime.AllocatorCounters, "counters.bin");
            Save(cmd, VirtualShadowMapPrototypeRuntime.Projections.Buffer, "projections.bin");
        }

        private static void ValidatePool(RenderTexture pool, int capacity, int row, int size)
        {
            if (pool == null || pool.dimension != TextureDimension.Tex2DArray || pool.graphicsFormat != GraphicsFormat.R32_UInt
                || pool.volumeDepth != VirtualShadowMapPrototypeRuntime.DepthLayerCount || pool.width < row * size
                || pool.height < ((capacity + row - 1) / row) * size)
                throw new InvalidOperationException("Capacity audit requires the complete current R32_UInt 16-layer pool.");
        }

        private void Save(CommandBuffer cmd, GraphicsBuffer buffer, string file)
        {
            string path = m_Directory + "/" + file;
            m_Pending++;
            cmd.RequestAsyncReadback(buffer, request => {
                try
                {
                    if (request.hasError) m_Error = "GPU readback failed: " + path;
                    else File.WriteAllBytes(path, request.GetData<byte>().ToArray());
                }
                catch (Exception exception) { m_Error = exception.ToString(); }
                finally { m_Pending--; }
            });
        }

        private void OnReload() => Finish("interrupted: assembly reload");
        private void OnQuit() => Finish("interrupted: editor quit");
        private void Finish(string status)
        {
            if (m_Disposed) return;
            try { Dispose(); }
            finally { File.WriteAllText(m_Root + "/status.txt", status); }
        }

        public void Dispose()
        {
            if (m_Disposed) return;
            m_Disposed = true;
            AssetDatabase.LoadAssetAtPath<ComputeShader>("Packages/com.vivid.render-pipelines/Shaders/Core/Private/CSMShadowResolve.compute").SetInt("_VSMPageOccupancySkipDisabled", 0);
            CSMShadowResolvePass.EditorReceiverCapture -= Capture;
            EditorApplication.update -= Tick;
            AssemblyReloadEvents.beforeAssemblyReload -= OnReload;
            EditorApplication.quitting -= OnQuit;
            try
            {
                if (m_Pending > 0) AsyncGPUReadback.WaitAllRequests();
                m_Timing?.Dispose(); m_Pages?.Dispose(); m_Summary?.Dispose();
                Object.DestroyImmediate(m_VolumeObject); Object.DestroyImmediate(m_Profile); Object.DestroyImmediate(m_Audit);
            }
            finally
            {
                Application.runInBackground = m_Background;
                s_Active = null;
                EditorApplication.QueuePlayerLoopUpdate();
            }
        }
    }
