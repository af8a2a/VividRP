#pragma kernel ResetComparison
#pragma kernel CompareResolve
Texture2D<float> _Before;
Texture2D<float> _After;
RWStructuredBuffer<uint> _Summary;
int _Width, _Height;
groupshared uint g_Different, g_Max, g_Shadowed;
[numthreads(1,1,1)]
void ResetComparison(uint3 id:SV_DispatchThreadID) { for(uint i=0;i<4;i++) _Summary[i]=0; }
[numthreads(16,16,1)]
void CompareResolve(uint3 id:SV_DispatchThreadID,uint3 group:SV_GroupID,uint lane:SV_GroupIndex)
{
    if(lane==0) { g_Different=0;g_Max=0;g_Shadowed=0; }
    GroupMemoryBarrierWithGroupSync();
    if(id.x < (uint)_Width && id.y < (uint)_Height) {
        float a=_Before.Load(int3(id.xy,0)),b=_After.Load(int3(id.xy,0));
        if(asuint(a)!=asuint(b) || !isfinite(a) || !isfinite(b)) { InterlockedAdd(g_Different,1);InterlockedMax(g_Max,asuint(abs(a-b))); }
        if(a < 0.999) InterlockedAdd(g_Shadowed,1);
    }
    GroupMemoryBarrierWithGroupSync();
    if(lane==0) { InterlockedAdd(_Summary[0],g_Different);InterlockedMax(_Summary[1],g_Max);InterlockedAdd(_Summary[2],min(16u,(uint)_Width-group.x*16u)*min(16u,(uint)_Height-group.y*16u));InterlockedAdd(_Summary[3],g_Shadowed); }
}
