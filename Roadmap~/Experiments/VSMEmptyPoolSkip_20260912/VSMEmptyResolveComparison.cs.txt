using System;
using System.IO;
using UnityEditor;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Experimental.Rendering;
namespace VividRP.Editor
{
    // Explicit diagnostic replay. Resolve reads the SAME current inputs and
    // previous history twice; only separate scratch output targets are written.
    internal sealed class VSMEmptyResolveComparison : IDisposable
    {
        private RenderTexture before, after;
        private ComputeShader resolve, compare;
        private GraphicsBuffer summary;
        private int resolveKernel, resetKernel, compareKernel, pending;
        internal VSMEmptyResolveComparison(int width, int height)
        {
            resolve=AssetDatabase.LoadAssetAtPath<ComputeShader>("Packages/com.vivid.render-pipelines/Shaders/Core/Private/CSMShadowResolve.compute");
            compare=AssetDatabase.LoadAssetAtPath<ComputeShader>("Packages/com.vivid.render-pipelines/Editor/Tools/VSMEmptyResolveCompare.compute");
            resolveKernel=resolve.FindKernel("CSMShadowResolve");resetKernel=compare.FindKernel("ResetComparison");compareKernel=compare.FindKernel("CompareResolve");
            var d=new RenderTextureDescriptor(width,height,GraphicsFormat.R32_SFloat,0) {enableRandomWrite=true,msaaSamples=1};
            before=new RenderTexture(d) {name="VSMEmptyComparisonBefore"};before.Create();
            after=new RenderTexture(d) {name="VSMEmptyComparisonAfter"};after.Create();
            summary=new GraphicsBuffer(GraphicsBuffer.Target.Structured,4,4);
        }
        internal void Record(CommandBuffer cmd, Texture originalOutput, string prefix)
        {
            resolve.GetKernelThreadGroupSizes(resolveKernel,out uint x,out uint y,out _);
            cmd.SetComputeIntParam(resolve,"_VSMPageOccupancySkipDisabled",1);
            cmd.SetComputeTextureParam(resolve,resolveKernel,"_DirectionalShadowTexture",before);
            cmd.DispatchCompute(resolve,resolveKernel,Mathf.CeilToInt(before.width/(float)x),Mathf.CeilToInt(before.height/(float)y),1);
            cmd.SetComputeIntParam(resolve,"_VSMPageOccupancySkipDisabled",0);
            cmd.SetComputeTextureParam(resolve,resolveKernel,"_DirectionalShadowTexture",after);
            cmd.DispatchCompute(resolve,resolveKernel,Mathf.CeilToInt(after.width/(float)x),Mathf.CeilToInt(after.height/(float)y),1);
            cmd.SetComputeTextureParam(resolve,resolveKernel,"_DirectionalShadowTexture",originalOutput);
            cmd.SetComputeBufferParam(compare,resetKernel,"_Summary",summary);cmd.DispatchCompute(compare,resetKernel,1,1,1);
            cmd.SetComputeBufferParam(compare,compareKernel,"_Summary",summary);
            cmd.SetComputeTextureParam(compare,compareKernel,"_Before",before);cmd.SetComputeTextureParam(compare,compareKernel,"_After",after);
            cmd.SetComputeIntParam(compare,"_Width",before.width);cmd.SetComputeIntParam(compare,"_Height",before.height);
            cmd.DispatchCompute(compare,compareKernel,(before.width+15)/16,(before.height+15)/16,1);
            pending++;
            cmd.RequestAsyncReadback(summary,r=>{
                try {
                    if(r.hasError) throw new InvalidOperationException("same-frame comparison readback failed");
                    var data=r.GetData<uint>();
                    File.WriteAllText(prefix+"_same-frame.json","{\"different_pixels\":"+data[0]+",\"max_abs_bits\":"+data[1]+",\"checked_pixels\":"+data[2]+",\"shadowed_pixels\":"+data[3]+"}");
                } finally {pending--;}
            });
        }
        public void Dispose()
        {
            if(pending!=0) AsyncGPUReadback.WaitAllRequests();
            summary?.Dispose();if(before!=null) {before.Release();UnityEngine.Object.DestroyImmediate(before);}if(after!=null) {after.Release();UnityEngine.Object.DestroyImmediate(after);}
        }
    }
}
