using System;
using System.IO;
using UnityEditor;
using UnityEngine;
using UnityEngine.Rendering;
using Unity.Mathematics;
using VividRP.Runtime;
using VividRP.Runtime.RenderPass.Core;
using Object = UnityEngine.Object;
namespace VividRP.Editor
{
    internal static class VSMCurrentFrameValidation
    {
        [Serializable] private sealed class Report
        {
            public int gpuCases, repeatedRequests, zeroFrameRequests;
        }
        private static void Check(bool ok, string text) { if (!ok) throw new InvalidOperationException(text); }
        [MenuItem("Tools/VividRP/Diagnostics/Validate Temporary VSM Current Frame")]
        internal static void Run()
        {
            Check(!EditorApplication.isPlaying, "Use Edit Mode for the isolated GPU diagnostic.");
            var report = new Report();
            for (int mode = 0; mode < 3; mode++) CheckGpu(mode, report);
            File.WriteAllText("Packages/VividRP/Temp~/smrt-path/marking.json", JsonUtility.ToJson(report, true));
            Debug.Log("VSM current-frame diagnostic passed: " + JsonUtility.ToJson(report));
        }
        private static void CheckGpu(int mode, Report report)
        {
            var shader = Object.Instantiate(PipelineResourceManager.Get<VividRPCoreResources>().CSMShadowResolveCompute);
            var depth = new Texture2D(8, 8, TextureFormat.RFloat, false, true);
            var normal = new Texture2D(8, 8, TextureFormat.RGBAFloat, false, true);
            using var projections = new GraphicsBuffer(GraphicsBuffer.Target.Structured, 3, 160);
            using var metadata = new GraphicsBuffer(GraphicsBuffer.Target.Structured, 12, 16);
            using var table = new GraphicsBuffer(GraphicsBuffer.Target.Structured, 12, 4);
            using var owners = new GraphicsBuffer(GraphicsBuffer.Target.Structured, 16, 4);
            using var counters = new GraphicsBuffer(GraphicsBuffer.Target.Structured, 4, 4);
            try
            {
                var records = new VirtualShadowMapProjection[3];
                for (int i = 0; i < 3; i++)
                {
                    var matrix = Matrix4x4.identity; matrix.m00 = matrix.m11 = .1f / (1 << i);
                    matrix.m03 = matrix.m13 = matrix.m23 = .5f;
                    records[i] = new VirtualShadowMapProjection { WorldToShadow = matrix,
                        SelectionSphere = new Vector4(0, 0, 0, -5 * (1 << i)), Parameters = new Vector4(1 << i, 0, .1f, 100) };
                }
                projections.SetData(records); metadata.SetData(new uint4[12]); table.SetData(new uint[12]); owners.SetData(new uint[16]);
                var pixels = new float[64]; var normals = new Color[64];
                for (int i = 0; i < 64; i++) { pixels[i] = .5f; normals[i] = new Color(.5f, .5f, 0, 0); }
                depth.SetPixelData(pixels, 0); depth.Apply(); normal.SetPixels(normals); normal.Apply();
                shader.SetInt("_CSMOutputWidth", 8); shader.SetInt("_CSMOutputHeight", 8);
                shader.SetInt("_VSMProjectionCount", 3); shader.SetInt("_VSMPrototypeVirtualResolution", 8);
                shader.SetInt("_VSMPrototypePageSize", 4); shader.SetInt("_VSMPrototypePagesPerAxis", 2);
                shader.SetInt("_VSMPrototypePageTableEntryCount", 12); shader.SetInt("_VSMPrototypePhysicalPageCapacity", 16);
                shader.SetInt("_VSMPrototypeRequestEnabled", 1);
                shader.SetMatrix("_CSMInvViewProjMatrix", Matrix4x4.identity); shader.SetMatrix("_VSMReceiverViewProjection", Matrix4x4.identity);
                shader.SetVector("_VSMReceiverParameters", Vector4.zero);
                shader.SetVector("_VSMReceiverQuality", new Vector4(mode > 0 ? 1 : 0, 1, .025f, 1));
                shader.SetVector("_VSMSMRTParameters", mode == 2 ? new Vector4(4, 4, 6, .5f) : Vector4.zero);
                int mark = shader.FindKernel("VSMMarkReceiverPages"), allocate = shader.FindKernel("VSMPrototypeAllocatePages");
                shader.SetBuffer(mark, "_VSMProjections", projections); shader.SetBuffer(mark, "_VSMPrototypePageMetadata", metadata);
                shader.SetTexture(mark, "_DepthTexture", depth); shader.SetTexture(mark, "_GBuffer1", normal);
                shader.SetBuffer(allocate, "_VSMPrototypePageMetadata", metadata); shader.SetBuffer(allocate, "_VSMPrototypeWritablePageTable", table);
                shader.SetBuffer(allocate, "_VSMPrototypePhysicalPageOwners", owners); shader.SetBuffer(allocate, "_VSMPrototypeAllocatorCounters", counters);
                var result = new uint[4]; var data = new uint4[12]; int count = 0;
                for (int frame = 0; frame < 3; frame++)
                {
                    if (frame == 2) { for (int i = 0; i < 64; i++) pixels[i] = SystemInfo.usesReversedZBuffer ? 0 : 1;
                        depth.SetPixelData(pixels, 0); depth.Apply(); }
                    int clear = shader.FindKernel(frame == 0 ? "VSMPrototypeResetReceiverFeedback" : "VSMPrototypeClearReceiverRequests");
                    shader.SetBuffer(clear, "_VSMPrototypePageMetadata", metadata); shader.Dispatch(clear, 1, 1, 1);
                    shader.SetInt("_CSMFrameIndex", frame); shader.SetInt("_VSMPrototypeFeedbackFrameIndex", frame);
                    shader.Dispatch(mark, 1, 1, 1);
                    if (mode == 2 && frame < 2)
                    {
                        metadata.GetData(data);
                        for (int page = 4; page < 12; page++) Check((data[page].x & 512u) != 0, "Continuation lost primary priority.");
                    }
                    shader.Dispatch(allocate, 1, 1, 1); counters.GetData(result);
                    if (frame == 0) { count = (int)result[1]; Check(count > 0 && result[2] == count && result[3] == 0, "Cold frame demand/allocate failed."); report.zeroFrameRequests += count; }
                    if (frame == 1) { Check(result[1] == count && result[2] == 0 && result[3] == 0, "Repeated requests failed."); report.repeatedRequests += (int)result[1]; }
                    if (frame == 2) { Check(result[1] == 0, "Sky retained stale demand."); metadata.GetData(data);
                        foreach (var m in data) if (m.y != 0) Check(m.z == 1, "Lost LRU age."); }
                }
                report.gpuCases++;
            }
            finally { Object.DestroyImmediate(shader); Object.DestroyImmediate(depth); Object.DestroyImmediate(normal); }
        }
    }
}
