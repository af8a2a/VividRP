using System;
using System.IO;
using System.IO.Compression;
using UnityEditor;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Experimental.Rendering;
using VividRP.Runtime;
using VividRP.Runtime.RenderPass.Core;
using Object = UnityEngine.Object;
namespace VividRP.Editor
{
    internal sealed class VSMNoiseSamplingProbe : IDisposable
    {
        static VSMNoiseSamplingProbe active;
        static bool nextTiming;
        readonly bool timing, background, recording;
        readonly Camera camera;
        readonly Vector3 position;
        readonly Quaternion rotation;
        readonly string root;
        readonly string[] names;
        readonly float[] times=new float[128];
        ComputeShader resolve, copyShader;
        RenderTexture sampled;
        int stage,warm,step,last=-1,pending,copy;
        bool disposed,started;
        double time;
        [Serializable] sealed class Frame {public int frame,phase,step,width,height;public Matrix4x4 vp;public Vector3 cameraPosition;}
        [MenuItem("Tools/VividRP/Diagnostics/Run VSM Noise Sampling")]
        static void Run()
        {
            if(active!=null)throw new InvalidOperationException("Active probe");
            Camera selected=null;foreach(var c in Object.FindObjectsByType<Camera>())if(c.cameraType==CameraType.Game&&c.isActiveAndEnabled){if(selected!=null)throw new InvalidOperationException("Multiple cameras");selected=c;}
            if(selected==null||EditorApplication.isPlayingOrWillChangePlaymode)throw new InvalidOperationException("Idle Game camera required");
            active=new VSMNoiseSamplingProbe(selected,nextTiming);nextTiming=false;
            try{active.Begin();}catch{active.Dispose();throw;}
        }
        [MenuItem("Tools/VividRP/Diagnostics/Time VSM Noise Sampling")]
        static void Time(){nextTiming=true;Run();}
        VSMNoiseSamplingProbe(Camera c,bool t)
        {
            camera=c;position=c.transform.position;rotation=c.transform.rotation;timing=t;background=Application.runInBackground;
            recording=UnityEngine.Profiling.Recorder.Get("VSM.Resolve").enabled;
            names=t?new[]{"baseline4_forward","clip4_forward","clip4_reverse","baseline4_reverse"}:new[]{"baseline4","clip4"};
            root=Path.GetFullPath("Packages/VividRP/Temp~/vsm-noise-stability/"+(t?"timing/":"sampling/")+DateTime.UtcNow.ToString("yyyyMMdd_HHmmss_fff"));
        }
        void Begin()
        {
            Directory.CreateDirectory(root);File.WriteAllText(root+"/settings.json",JsonUtility.ToJson(VividVolumeManagerUtility.GetCascadedShadowSettingsVolume()));
            File.WriteAllText(root+"/camera.json",EditorJsonUtility.ToJson(camera));File.WriteAllText(root+"/transform.json",EditorJsonUtility.ToJson(camera.transform));
            resolve=AssetDatabase.LoadAssetAtPath<ComputeShader>("Packages/com.vivid.render-pipelines/Shaders/Core/Private/CSMShadowResolve.compute");
            copyShader=Object.Instantiate(AssetDatabase.LoadAssetAtPath<ComputeShader>("Packages/com.vivid.render-pipelines/Editor/Tools/VSMNoiseTemporalProbe.compute"));copy=copyShader.FindKernel("CopyStabilitySignal");
            if(!timing){sampled=new RenderTexture(new RenderTextureDescriptor(240,180,GraphicsFormat.R32G32B32A32_SFloat,0){enableRandomWrite=true});sampled.Create();}
            VSMProfiling.Resolve.enableRecording=true;Application.runInBackground=true;CSMShadowResolvePass.EditorReceiverCapture+=Capture;
            EditorApplication.update+=Tick;AssemblyReloadEvents.beforeAssemblyReload+=Dispose;Start();
        }
        void Start(){warm=step=0;started=false;time=EditorApplication.timeSinceStartup;Directory.CreateDirectory(root+"/"+names[stage]);File.WriteAllText(root+"/progress.txt",names[stage]);resolve.SetInt("_VSMNoiseStrictHistoryClip",names[stage].StartsWith("clip4")?0:1);camera.GetComponent<VividAdditionalCameraData>().ResetPostProcessingHistory();}
        void Tick()
        {
            if(disposed)return;
            if(camera.transform.position!=position||camera.transform.rotation!=rotation||EditorApplication.timeSinceStartup-time>180){File.WriteAllText(root+"/status.txt","failed: movement or timeout");Dispose();return;}
            if(step>=(timing?128:256)&&pending==0){if(timing)File.WriteAllLines(root+"/"+names[stage]+"/gpu-ms.txt",Array.ConvertAll(times,x=>x.ToString("R",System.Globalization.CultureInfo.InvariantCulture)));if(++stage==names.Length){File.WriteAllText(root+"/status.txt","complete");Dispose();return;}Start();}
            EditorApplication.QueuePlayerLoopUpdate();UnityEditorInternal.InternalEditorUtility.RepaintAllViews();
        }
        void Capture(ComputePassContext context,Texture device,Texture normal,Texture shadow)
        {
            var c=context.Get<VividCameraData>();if(disposed||c.camera!=camera||c.frameIndex==last||step>=(timing?128:256))return;last=c.frameIndex;
            if(!VirtualShadowMapPrototypeRuntime.IsFrameActive)throw new InvalidOperationException("VSM inactive");
            if(++warm<=64)return;if(!started){if(!timing&&(c.frameIndex&255)!=0)return;started=true;}
            if(timing){times[step++]=VSMProfiling.Resolve.gpuElapsedTime;return;}
            if((c.frameIndex&255)!=step)throw new InvalidOperationException("Non-consecutive phase");
            if(step==0)camera.GetComponent<VividAdditionalCameraData>().RequestFinalFrameScreenshot(root+"/"+names[stage]+"/scene.png");
            string prefix=root+"/"+names[stage]+"/frame_"+step.ToString("D3");
            File.WriteAllText(prefix+".json",JsonUtility.ToJson(new Frame{frame=c.frameIndex,phase=c.frameIndex&255,step=step,width=device.width,height=device.height,vp=c.GetGPUViewProjectionMatrix(true),cameraPosition=camera.transform.position}));
            if(!camera.GetVividCameraHistory().TryGetTexture(CameraHistoryId.Create("VSMShadowHistory"),out var history)||!history.IsValid())throw new InvalidOperationException("No history");
            var cmd=context.cmd.m_WrappedCommandBuffer;cmd.SetComputeIntParams(copyShader,"_StabilityOrigin",482,162);cmd.SetComputeTextureParam(copyShader,copy,"_VSMHistoryCurrent",history.GetFrame(0).rt);cmd.SetComputeTextureParam(copyShader,copy,"_StabilitySampled",sampled);cmd.DispatchCompute(copyShader,copy,30,23,1);
            pending++;cmd.RequestAsyncReadback(sampled,0,TextureFormat.RGBAFloat,r=>{try{if(r.hasError)throw new IOException("readback failed");var bytes=r.GetData<byte>().ToArray();using(var f=File.Create(prefix+".gz"))using(var z=new GZipStream(f,System.IO.Compression.CompressionLevel.Fastest))z.Write(bytes,0,bytes.Length);}finally{pending--;}});step++;
        }
        public void Dispose(){if(disposed)return;disposed=true;CSMShadowResolvePass.EditorReceiverCapture-=Capture;EditorApplication.update-=Tick;AssemblyReloadEvents.beforeAssemblyReload-=Dispose;AsyncGPUReadback.WaitAllRequests();resolve?.SetInt("_VSMNoiseStrictHistoryClip",0);if(sampled!=null){sampled.Release();Object.DestroyImmediate(sampled);}Object.DestroyImmediate(copyShader);Application.runInBackground=background;VSMProfiling.Resolve.enableRecording=recording;active=null;EditorApplication.QueuePlayerLoopUpdate();}
    }
}
