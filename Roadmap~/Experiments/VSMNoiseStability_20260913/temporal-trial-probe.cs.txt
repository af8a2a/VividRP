using System;
using System.IO;
using System.IO.Compression;
using UnityEditor;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Experimental.Rendering;
using VividRP.Runtime;
using VividRP.Runtime.RenderPass.Core;
using Object = UnityEngine.Object;

namespace VividRP.Editor
{
    // Temporary same-input temporal comparison; no production history writes.
    internal sealed class VSMNoiseStabilityProbe : IDisposable
    {
        static VSMNoiseStabilityProbe active;
        readonly Camera camera;
        readonly Light light;
        readonly Vector3 position;
        readonly Quaternion rotation, lightRotation;
        readonly bool background;
        readonly string root;
        readonly string[] names = { "baseline4", "cap8", "responsive8" };
        readonly string[] scenarios = { "static", "camera_slide", "sun_step" };
        readonly RenderTexture[,] histories = new RenderTexture[3,2];
        readonly RenderTexture[,] depths = new RenderTexture[3,2];
        readonly RenderTexture[] outputs = new RenderTexture[3];
        RenderTexture sampled;
        ComputeShader shader;
        Matrix4x4 previousVP, previousView;
        Vector3 previousLight;
        int kernel, copy, frame, stage, lastFrame=-1, pending, width, height;
        bool disposed;
        double started;
        [Serializable] sealed class Frame
        {
            public int frame, phase, step, width, height;
            public Matrix4x4 viewProjection;
            public Vector3 cameraPosition, lightDirection;
            public bool historyValid;
        }
        [MenuItem("Tools/VividRP/Diagnostics/Run VSM Noise Stability")]
        static void Run()
        {
            if(active!=null) throw new InvalidOperationException("Noise probe active");
            if(EditorApplication.isPlayingOrWillChangePlaymode) throw new InvalidOperationException("Edit Mode required");
            Camera c=null;Light l=null;
            foreach(var x in Object.FindObjectsByType<Camera>()) if(x.cameraType==CameraType.Game&&x.isActiveAndEnabled) {if(c!=null)throw new InvalidOperationException("Multiple Game cameras");c=x;}
            foreach(var x in Object.FindObjectsByType<Light>()) if(x.type==LightType.Directional&&x.isActiveAndEnabled) {if(l!=null)throw new InvalidOperationException("Multiple suns");l=x;}
            if(c==null||l==null)throw new InvalidOperationException("Missing camera or sun");
            active=new VSMNoiseStabilityProbe(c,l);
            try{active.Begin();}catch{active.Dispose();throw;}
        }
        VSMNoiseStabilityProbe(Camera c,Light l)
        {
            camera=c;light=l;position=c.transform.position;rotation=c.transform.rotation;lightRotation=l.transform.rotation;
            background=Application.runInBackground;
            root=Path.GetFullPath("Packages/VividRP/Temp~/vsm-noise-stability/replay/"+DateTime.UtcNow.ToString("yyyyMMdd_HHmmss_fff"));
        }
        void Begin()
        {
            Directory.CreateDirectory(root);
            File.WriteAllText(root+"/before-camera.json",EditorJsonUtility.ToJson(camera));
            File.WriteAllText(root+"/before-camera-transform.json",EditorJsonUtility.ToJson(camera.transform));
            File.WriteAllText(root+"/before-light-transform.json",EditorJsonUtility.ToJson(light.transform));
            File.WriteAllText(root+"/settings.json",JsonUtility.ToJson(VividVolumeManagerUtility.GetCascadedShadowSettingsVolume()));
            File.WriteAllText(root+"/environment.txt",Application.unityVersion+" "+SystemInfo.graphicsDeviceName+" "+SystemInfo.graphicsDeviceType+" background="+background);
            var scenes=new System.Text.StringBuilder();for(int i=0;i<UnityEngine.SceneManagement.SceneManager.sceneCount;i++){var s=UnityEngine.SceneManagement.SceneManager.GetSceneAt(i);scenes.AppendLine(s.name+" dirty="+s.isDirty);}
            File.WriteAllText(root+"/before-scenes.txt",scenes.ToString());
            shader=Object.Instantiate(AssetDatabase.LoadAssetAtPath<ComputeShader>("Packages/com.vivid.render-pipelines/Editor/Tools/VSMNoiseTemporalProbe.compute"));
            kernel=shader.FindKernel("VSMShadowTemporalV");copy=shader.FindKernel("CopyStabilitySignal");
            CSMShadowResolvePass.EditorReceiverCapture+=Capture;EditorApplication.update+=Tick;AssemblyReloadEvents.beforeAssemblyReload+=Dispose;
            Application.runInBackground=true;StartStage();
        }
        void StartStage()
        {
            frame=0;started=EditorApplication.timeSinceStartup;
            camera.transform.SetPositionAndRotation(position,rotation);light.transform.rotation=lightRotation;
            Directory.CreateDirectory(root+"/"+scenarios[stage]);File.WriteAllText(root+"/progress.txt",scenarios[stage]);
        }
        void Tick()
        {
            if(disposed)return;
            if(EditorApplication.timeSinceStartup-started>180){File.WriteAllText(root+"/status.txt","timeout");Dispose();return;}
            if(frame>=192&&pending==0){if(++stage==scenarios.Length){File.WriteAllText(root+"/status.txt","complete");Dispose();return;}StartStage();}
            EditorApplication.QueuePlayerLoopUpdate();UnityEditorInternal.InternalEditorUtility.RepaintAllViews();
        }
        RenderTexture Make(GraphicsFormat format,int w,int h)
        {
            var t=new RenderTexture(new RenderTextureDescriptor(w,h,format,0){enableRandomWrite=true,msaaSamples=1}){hideFlags=HideFlags.HideAndDontSave};t.Create();return t;
        }
        void Capture(ComputePassContext context,Texture device,Texture normal,Texture productionShadow)
        {
            var c=context.Get<VividCameraData>();if(disposed||c.camera!=camera||c.frameIndex==lastFrame||frame>=192)return;lastFrame=c.frameIndex;
            if(!VirtualShadowMapPrototypeRuntime.IsFrameActive)throw new InvalidOperationException("VSM inactive");
            var cmd=context.cmd.m_WrappedCommandBuffer;
            if(sampled==null){width=device.width;height=device.height;sampled=Make(GraphicsFormat.R32G32B32A32_SFloat,240,180);for(int v=0;v<3;v++){outputs[v]=Make(GraphicsFormat.R16_SFloat,width,height);for(int p=0;p<2;p++){histories[v,p]=Make(GraphicsFormat.R16G16B16A16_SFloat,width,height);depths[v,p]=Make(GraphicsFormat.R32_SFloat,width,height);}}}
            var vp=c.GetGPUViewProjectionMatrix(true);var view=camera.worldToCameraMatrix;var direction=-light.transform.forward;
            bool valid=frame>0&&direction==previousLight;
            int dst=frame&1,src=dst^1;
            string prefix=root+"/"+scenarios[stage]+"/frame_"+(frame-64).ToString("D3");
            if(frame>=64)File.WriteAllText(prefix+".json",JsonUtility.ToJson(new Frame{frame=c.frameIndex,phase=c.frameIndex&255,step=frame-64,width=width,height=height,viewProjection=vp,cameraPosition=camera.transform.position,lightDirection=direction,historyValid=valid}));
            cmd.SetComputeIntParam(shader,"_CSMOutputWidth",width);cmd.SetComputeIntParam(shader,"_CSMOutputHeight",height);
            cmd.SetComputeMatrixParam(shader,"_CSMInvViewProjMatrix",vp.inverse);cmd.SetComputeMatrixParam(shader,"_VSMPreviousViewProjection",previousVP);
            cmd.SetComputeMatrixParam(shader,"_VSMPreviousView",previousView);cmd.SetComputeMatrixParam(shader,"_VSMCurrentView",view);
            cmd.SetComputeTextureParam(shader,kernel,"_DepthTexture",device);cmd.SetComputeTextureParam(shader,kernel,"_GBuffer1",normal);
            cmd.SetComputeTextureParam(shader,kernel,"_CSMShadowFilterSource",CSMShadowResolvePass.EditorNoiseHorizontalSource);
            cmd.SetComputeIntParams(shader,"_StabilityOrigin",482,162);
            for(int v=0;v<3;v++)
            {
                cmd.SetComputeVectorParam(shader,"_VSMHistoryParameters",new Vector4(valid?1:0,0,v==0?4:8,v==2?1:0));
                cmd.SetComputeTextureParam(shader,kernel,"_VSMHistoryPrevious",histories[v,src]);cmd.SetComputeTextureParam(shader,kernel,"_VSMDepthPrevious",depths[v,src]);
                cmd.SetComputeTextureParam(shader,kernel,"_VSMHistoryCurrent",histories[v,dst]);cmd.SetComputeTextureParam(shader,kernel,"_VSMDepthCurrent",depths[v,dst]);
                cmd.SetComputeTextureParam(shader,kernel,"_CSMShadowFilterTexture",outputs[v]);cmd.DispatchCompute(shader,kernel,(width+7)/8,(height+7)/8,1);
                if(frame>=64){cmd.SetComputeTextureParam(shader,copy,"_VSMHistoryCurrent",histories[v,dst]);cmd.SetComputeTextureParam(shader,copy,"_StabilitySampled",sampled);cmd.DispatchCompute(shader,copy,30,23,1);Save(cmd,sampled,prefix+"_"+names[v]+".gz");}
            }
            if(frame>=64)
            {
                var history=camera.GetVividCameraHistory();if(history.TryGetTexture(CameraHistoryId.Create("VSMShadowHistory"),out var h)&&h.IsValid())
                {cmd.SetComputeTextureParam(shader,copy,"_VSMHistoryCurrent",h.GetFrame(0).rt);cmd.SetComputeTextureParam(shader,copy,"_StabilitySampled",sampled);cmd.DispatchCompute(shader,copy,30,23,1);Save(cmd,sampled,prefix+"_production.gz");}
            }
            previousVP=vp;previousView=view;previousLight=direction;frame++;
            int step=frame-64;
            if(stage==1&&step>=0)camera.transform.position=position+camera.transform.right*(.0025f*Mathf.Min(step,64));
            if(stage==2&&step>=0)light.transform.rotation=Quaternion.AngleAxis(step>=32&&step<80?1f:0f,Vector3.up)*lightRotation;
        }
        void Save(CommandBuffer cmd,Texture texture,string path)
        {
            pending++;cmd.RequestAsyncReadback(texture,0,TextureFormat.RGBAFloat,r=>{try{if(r.hasError)throw new IOException("readback failed");using(var f=File.Create(path))using(var z=new GZipStream(f,System.IO.Compression.CompressionLevel.Fastest))z.Write(r.GetData<byte>().ToArray(),0,r.GetData<byte>().Length);}finally{pending--;}});
        }
        public void Dispose()
        {
            if(disposed)return;disposed=true;CSMShadowResolvePass.EditorReceiverCapture-=Capture;EditorApplication.update-=Tick;AssemblyReloadEvents.beforeAssemblyReload-=Dispose;
            AsyncGPUReadback.WaitAllRequests();camera.transform.SetPositionAndRotation(position,rotation);light.transform.rotation=lightRotation;
            foreach(var x in histories)if(x!=null){x.Release();Object.DestroyImmediate(x);}foreach(var x in depths)if(x!=null){x.Release();Object.DestroyImmediate(x);}foreach(var x in outputs)if(x!=null){x.Release();Object.DestroyImmediate(x);}
            if(sampled!=null){sampled.Release();Object.DestroyImmediate(sampled);}Object.DestroyImmediate(shader);Application.runInBackground=background;
            File.WriteAllText(root+"/after-camera-transform.json",EditorJsonUtility.ToJson(camera.transform));File.WriteAllText(root+"/after-light-transform.json",EditorJsonUtility.ToJson(light.transform));
            active=null;EditorApplication.QueuePlayerLoopUpdate();
        }
    }
}
