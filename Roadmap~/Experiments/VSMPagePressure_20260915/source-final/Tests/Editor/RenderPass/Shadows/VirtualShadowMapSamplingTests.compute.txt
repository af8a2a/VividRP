// Same-input oracle for the ordered lookup, including independent pool ranks.
[numthreads(64, 1, 1)]
void InspectSMRTDepthIntervals(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    float4 input = _SamplingInputs[id.x], ray = _SamplingNormals[id.x];
    int2 texel = int2(input.xy);
    bool referenceHit = false;
    for (uint layer = 0; layer < VIVID_VSM_DEPTH_LAYER_COUNT; layer++)
    {
        uint2 depth = LoadVSMDepthLayer(texel, layer, 0u);
        float2 surface = float2(depth.x == 0u ? -1 : (asfloat(depth.x) - ray.x) / ray.y,
            depth.y == 0u ? -1 : (asfloat(depth.y) - ray.x) / ray.y);
        if (all(surface <= input.z)) break;
        if (any((surface > input.z) & (surface - ray.z <= input.w))) { referenceHit = true; break; }
    }
    uint2 front = LoadVSMDepthLayer(texel, 0u, 0u);
    bool ordered = VSMSMRTHasDepthInInterval(_VSMPrototypeStaticPhysicalPage,
        texel, front.x, ray.x, ray.y, input.z, input.w, ray.z);
    if (!ordered) ordered = VSMSMRTHasDepthInInterval(_VSMPrototypeDynamicPhysicalPage,
        texel, front.y, ray.x, ray.y, input.z, input.w, ray.z);
    _SamplingResults[id.x] = float2(referenceHit, ordered);
}

Texture2D<float> _SMRTBefore, _SMRTAfter;
RWStructuredBuffer<uint> _SMRTComparison;
int2 _SMRTComparisonSize;
groupshared uint g_SMRTDifferent, g_SMRTMax, g_SMRTShadowed;
[numthreads(1, 1, 1)]
void ResetSMRTComparison(uint3 id : SV_DispatchThreadID)
{
    for (uint i = 0; i < 4; i++) _SMRTComparison[i] = 0;
}
[numthreads(16, 16, 1)]
void CompareSMRTOutputs(uint3 id : SV_DispatchThreadID, uint3 group : SV_GroupID, uint lane : SV_GroupIndex)
{
    if (lane == 0u) { g_SMRTDifferent = 0; g_SMRTMax = 0; g_SMRTShadowed = 0; }
    GroupMemoryBarrierWithGroupSync();
    if (all(id.xy < (uint2)_SMRTComparisonSize))
    {
        float a = _SMRTBefore.Load(int3(id.xy, 0)), b = _SMRTAfter.Load(int3(id.xy, 0));
        if (asuint(a) != asuint(b) || !isfinite(a) || !isfinite(b))
        { InterlockedAdd(g_SMRTDifferent, 1); InterlockedMax(g_SMRTMax, asuint(abs(a - b))); }
        if (a < 0.999) InterlockedAdd(g_SMRTShadowed, 1);
    }
    GroupMemoryBarrierWithGroupSync();
    if (lane == 0u)
    {
        InterlockedAdd(_SMRTComparison[0], g_SMRTDifferent);
        InterlockedMax(_SMRTComparison[1], g_SMRTMax);
        InterlockedAdd(_SMRTComparison[2], min(16u, (uint)_SMRTComparisonSize.x - group.x * 16u)
            * min(16u, (uint)_SMRTComparisonSize.y - group.y * 16u));
        InterlockedAdd(_SMRTComparison[3], g_SMRTShadowed);
    }
}

// input: pixel.xy, frame, mode (disk phase / receiver phase / disk / receiver).
// normal.xy carries ray index and ray count.
[numthreads(64, 1, 1)]
void InspectSMRTSamples(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    float4 input = _SamplingInputs[id.x];
    uint mode = (uint)input.w;
    float2 phase = VSMSMRTPhase(uint2(input.xy), (uint)input.z, (mode & 1u) * 2u);
    uint2 ray = uint2(_SamplingNormals[id.x].xy);
    _SamplingResults[id.x] = mode < 2u ? phase : mode == 2u
        ? VSMSMRTDiskSample(phase, ray.x, ray.y) : VSMSMRTReceiverOffset(phase, ray.x, ray.y);
}

[numthreads(8, 8, 1)]
void UploadTestPools(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= 16 || id.y >= 16) return;
    uint index = id.z * 256 + id.y * 16 + id.x;
    _TestStaticPool[id] = _TestStaticData[index];
    _TestDynamicPool[id] = _TestDynamicData[index];
}

#if defined(VIVID_VSM_RECEIVER_DEBUG)

[numthreads(64, 1, 1)]
void InspectReceiverDiagnostics(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    g_VSMDebugLevels = -1;
    g_VSMDebugBlend = 0;
    g_VSMDebugWork = 0;
    g_VSMDebugMissing = 0;
    g_VSMDebugQuality = -1;
    float shadow = ResolveVSMReceiver(_SamplingInputs[id.x].xyz, _SamplingNormals[id.x].xyz, uint2(_SamplingPixel));
    _DiagnosticResults[id.x] = _VSMReceiverDebugMode == 6
        ? (_VSMReceiverQuality.x > 0 ? g_VSMDebugQuality : -1) : _VSMReceiverDebugMode == 4
        ? float4(g_VSMDebugWork, g_VSMDebugLevels.z >= 0 ? 1 : 0)
        : _VSMReceiverDebugMode == 5 ? float4(g_VSMDebugMissing, shadow, 0, 0)
        : float4(g_VSMDebugLevels, g_VSMDebugBlend);
}

[numthreads(64, 1, 1)]
void SampleTaps(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    float4 input = _SamplingInputs[id.x];
    float shadow;
    bool valid = TrySampleVSMVirtualTap(input.xy, _SamplingOffsets[id.x], input.z, (int)input.w, shadow);
    _SamplingResults[id.x] = float2(valid ? 1.0 : 0.0, shadow);
}

[numthreads(64, 1, 1)]
void ResolveReceivers(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    _SamplingResults[id.x] = float2(
        ResolveVSMReceiver(_SamplingInputs[id.x].xyz, _SamplingNormals[id.x].xyz, uint2(_SamplingPixel)), 0.0);
}

[numthreads(64, 1, 1)]
void FilterFootprints(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    float4 input = _SamplingInputs[id.x];
    float shadow;
    bool valid = TryFilterVSMProjection(input.xyz, _SamplingNormals[id.x], (int)input.w, uint2(_SamplingPixel), shadow);
    _SamplingResults[id.x] = float2(valid ? 1.0 : 0.0, shadow);
}

[numthreads(64, 1, 1)]
void InspectVSMStochasticSample(uint3 id : SV_DispatchThreadID)

}

[numthreads(64, 1, 1)]
void InspectVSMStochasticTexelOffset(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;

[numthreads(64, 1, 1)]
void InspectBias(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    float4 input = _SamplingInputs[id.x];
    float3 normal = _SamplingNormals[id.x].xyz;
    normal *= rsqrt(max(dot(normal, normal), 1e-8));
    float4 bias = BuildVSMReceiverBias(_VSMProjections[(int)input.w], normal);
    _SamplingResults[id.x] = input.z > 0.0 ? bias.xy : bias.zw;
}

[numthreads(64, 1, 1)]
void InspectTransition(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    float4 input = _SamplingInputs[id.x];
    _SamplingResults[id.x] = float2(VSMTransitionWeight(input.x, input.y), 0.0);
}

[numthreads(64, 1, 1)]
void InspectScreenNormal(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    float4 input = _SamplingInputs[id.x];
    uint2 pixel = uint2(input.xy);
    float depth = _DepthTexture.Load(int3(pixel, 0));
    float3 normal = ReconstructVSMReceiverNormal(pixel, depth, ReconstructWorldPosition(pixel, depth),
        _SamplingNormals[id.x].xyz);
    _SamplingResults[id.x] = input.z > 0.0 ? float2(normal.z, 0.0) : normal.xy;
}

[numthreads(64, 1, 1)]
void ResolveScreenReceivers(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    uint2 pixel = uint2(_SamplingInputs[id.x].xy);
    float depth = _DepthTexture.Load(int3(pixel, 0));
    float3 position = ReconstructWorldPosition(pixel, depth);
    float3 normal = ReconstructVSMReceiverNormal(pixel, depth, position, _SamplingNormals[id.x].xyz);
    _SamplingResults[id.x] = float2(ResolveVSMReceiver(position, normal, pixel), 0.0);
}

[numthreads(64, 1, 1)]
void MarkFootprints(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    float4 input = _SamplingInputs[id.x];
    // z carries the primary/transition/parent role bits for shared-page feedback checks.
    MarkVSMReceiverPage(input.xy, (int)input.w, (uint)input.z);
}

[numthreads(64, 1, 1)]
void TraceSMRTRays(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    float4 input = _SamplingInputs[id.x];
    float4 ray = _SamplingNormals[id.x];
    float visibility;
    bool valid = TryTraceVSMSMRTRay(input.xyz, ray.xy, 0.05, ray.z,
        ray.w, (int)_VSMSMRTParameters.y, (int)input.w, visibility);
    _SamplingResults[id.x] = float2(valid ? 1 : 0, visibility);
}

[numthreads(64, 1, 1)]
void FilterSMRTFootprints(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    float4 input = _SamplingInputs[id.x];
    float visibility;
    bool valid = TryFilterVSMSMRT(input.xyz, _SamplingNormals[id.x], (int)input.w,
        uint2(id.x, _SamplingPixel.y), visibility);
    _SamplingResults[id.x] = float2(valid ? 1 : 0, visibility);
}

[numthreads(64, 1, 1)]
void TraceSMRTClipmaps(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    float4 input = _SamplingInputs[id.x];

    bool valid = TryTraceVSMSMRTClipmaps(input.xyz, _SamplingNormals[id.x].xy,
        0.05, (int)_VSMSMRTParameters.y, (int)input.w, visibility);

}

// Exact role/age union under divergent and partially active waves.
[numthreads(64, 1, 1)]
void MarkCoalescedFootprints(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount || _SamplingNormals[id.x].y == 0) return;
    float4 input = _SamplingInputs[id.x];
    MarkVSMReceiverPage(input.xy, (int)input.z, (uint)input.w, (int)_SamplingNormals[id.x].x);
}

// Reprojection and adaptive policy run through the shipping helpers.
[numthreads(64, 1, 1)]
void InspectVSMHistory(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    float3 world = _SamplingInputs[id.x].xyz;
    float3 normal = _SamplingNormals[id.x].xyz;
    uint2 pixel = uint2(_SamplingPixel);
    float2 history;
    bool valid = LoadVSMHistory(pixel, world, normal, history);
    _SamplingResults[id.x] = float2(valid ? history.x : -1, VSMAdaptiveRayCount(pixel, world, normal));
}

[numthreads(64, 1, 1)]
void InspectPressureDensity(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_SamplingCount) return;
    float blend;
    int level = SelectVSMDensityLevel(_SamplingInputs[id.x].xyz, _SamplingNormals[id.x].xyz, false, blend);
    _SamplingResults[id.x] = float2(level, blend);
}
