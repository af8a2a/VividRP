#pragma kernel Clear
#pragma kernel ClearCached
#pragma kernel Merge
RWTexture2DArray<uint> _Static;
RWTexture2DArray<uint> _Dynamic;
RWStructuredBuffer<uint> _Hits;
int _Dirty;
[numthreads(8,8,1)]
void Clear(uint3 id:SV_DispatchThreadID) {_Static[id]=0;_Dynamic[id]=0;}
[numthreads(8,8,1)]
void ClearCached(uint3 id:SV_DispatchThreadID) {if(_Dirty)_Static[id]=0;_Dynamic[id]=0;}
[numthreads(8,8,1)]
void Merge(uint3 id:SV_DispatchThreadID) {
 bool hit=false;
 for(uint layer=0;layer<16;layer++) {
  float2 d=float2(asfloat(_Static[uint3(id.xy,layer)]),asfloat(_Dynamic[uint3(id.xy,layer)]));
  hit=hit||any((d>.04)&(d<=.08));
 }
 _Hits[id.y*16+id.x]=hit;
}
