using System;
using System.IO;
using System.IO.Compression;
using UnityEditor;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Experimental.Rendering;
using VividRP.Runtime;
using VividRP.Runtime.GPUDriven;
using VividRP.Runtime.RenderPass.Core;
using Object = UnityEngine.Object;
namespace VividRP.Editor
{
    // Explicit live-scene diagnostic; no Unity Test Framework invocation.
    internal sealed class VSMResidualProbe : IDisposable
    {
        private static VSMResidualProbe active;
        private static readonly string[] DefaultCases = { "all8", "all10", "all12", "all16" };
        private readonly Camera camera;
        private bool variants;
        private string[] Cases;
        private readonly VividAdditionalCameraData additional;
        private readonly string root;
        private readonly bool background;
        private readonly Vector3 position;
        private readonly Quaternion rotation;
        private GameObject volumeObject;
        private VolumeProfile profile;
        private ComputeShader audit;
        private RenderTexture data, output, readbackDepth, readbackNormal, backDepth;
        private RayTracingAccelerationStructure acceleration;
        private GraphicsBuffer rays;
        private bool built;
        private int kernel, stage, warm, step, pending, lastFrame = -1;
        private double started;
        private string directory, error;
        private bool disposed;
        [Serializable] private sealed class Frame {
            public int width, height, frame, step, pages, levels;
            public Matrix4x4 viewProjection;
            public Vector3 cameraPosition, cameraEuler, lightDirection;
            public Vector4 receiver, smrt, quality;
            public int sampleIndexOffset;
            public string state, fallback, aa;
        }
        [MenuItem("Tools/VividRP/Diagnostics/Capture Temporary VSM Residual")]
        private static void Run() {
            if (active != null) throw new InvalidOperationException("Capture already active.");
            Camera selected = null;
            foreach (var candidate in Object.FindObjectsByType<Camera>(FindObjectsSortMode.None))
                if (candidate.cameraType == CameraType.Game && candidate.isActiveAndEnabled && candidate.gameObject.scene.IsValid()) {
                    if (selected != null) throw new InvalidOperationException("Multiple active Game cameras."); selected = candidate;
                }
            if (selected == null) throw new InvalidOperationException("No active Game camera.");
            active = new VSMResidualProbe(selected);
            try { active.Begin(); } catch { active.Dispose(); active = null; throw; }
        }
        private VSMResidualProbe(Camera c) { Cases=DefaultCases;
            camera = c; additional = c.GetComponent<VividAdditionalCameraData>();
            position = c.transform.position; rotation = c.transform.rotation; background = Application.runInBackground;
            root = Path.GetFullPath("Packages/VividRP/Temp~/vsm-residual/captures/" + DateTime.UtcNow.ToString("yyyyMMdd_HHmmss_fff"));
        }
        private void Begin() {
            Directory.CreateDirectory(root);
            File.WriteAllText(root + "/camera.json", EditorJsonUtility.ToJson(camera));
            File.WriteAllText(root + "/additional.json", EditorJsonUtility.ToJson(additional));
            File.WriteAllText(root + "/original-settings.json", JsonUtility.ToJson(VividVolumeManagerUtility.GetCascadedShadowSettingsVolume()));
            audit = Object.Instantiate(AssetDatabase.LoadAssetAtPath<ComputeShader>("Packages/com.vivid.render-pipelines/Editor/Tools/VSMResidualAudit.compute"));
            kernel = audit.FindKernel("VSMGeneralCoverageAudit");
            acceleration=new RayTracingAccelerationStructure(new RayTracingAccelerationStructure.Settings(RayTracingAccelerationStructure.ManagementMode.Manual,RayTracingAccelerationStructure.RayTracingModeMask.Everything,-1));
            foreach(var r in Object.FindObjectsByType<MeshletRenderer>()) {
                if(!r.isActiveAndEnabled||!r.sourceRenderingEnabled||r.shadowCastingMode==ShadowCastingMode.Off||r.sourceMesh==null)continue;
                for(int sub=0;sub<r.sourceMesh.subMeshCount;sub++) {
                    var cfg=new RayTracingMeshInstanceConfig{mesh=r.sourceMesh,meshLod=-1,subMeshIndex=(uint)sub,material=r.sourceMaterials[sub],mask=255,enableTriangleCulling=true,subMeshFlags=RayTracingSubMeshFlags.Enabled|RayTracingSubMeshFlags.ClosestHitOnly};
                    acceleration.AddInstance(in cfg,r.transform.localToWorldMatrix,null,0);
                }
            }
            foreach(var r in Object.FindObjectsByType<MeshRenderer>()) {
                if(!r.enabled||!r.gameObject.activeInHierarchy||r.shadowCastingMode==ShadowCastingMode.Off)continue;
                var m=r.GetComponent<MeshletRenderer>();if(m!=null&&m.isActiveAndEnabled&&m.takeOverSourceRenderer)continue;
                if(r.GetComponent<MeshFilter>()?.sharedMesh==null)continue;
                acceleration.AddInstance(r,null,true,false,255);
            }
            File.WriteAllText(root+"/geometry.txt","instances="+acceleration.GetInstanceCount());
            rays=new GraphicsBuffer(GraphicsBuffer.Target.Structured,1+64*16,16);
            volumeObject = new GameObject("Temporary VSM Residual") { hideFlags = HideFlags.HideAndDontSave };
            var volume = volumeObject.AddComponent<Volume>(); volume.isGlobal = true; volume.priority = float.MaxValue;
            Application.runInBackground = true;
            CSMShadowResolvePass.EditorReceiverCapture += Capture;
            EditorApplication.update += Tick; AssemblyReloadEvents.beforeAssemblyReload += Dispose;
            StartStage();
        }
        private void StartStage() {
            if (profile != null) Object.DestroyImmediate(profile);
            profile = ScriptableObject.CreateInstance<VolumeProfile>(); profile.hideFlags = HideFlags.HideAndDontSave;
            var settings = profile.Add<CascadedShadowSettingsVolume>();
            settings.screenSpaceShadowDenoise.Override(stage == 1);
            volumeObject.GetComponent<Volume>().sharedProfile = profile;
            directory = root + "/" + Cases[stage]; Directory.CreateDirectory(directory);
            warm = step = 0; started = EditorApplication.timeSinceStartup;
            File.WriteAllText(root + "/progress.txt", Cases[stage]);
        }
        private void Tick() {
            if (disposed) return;
            if (camera == null || camera.transform.position != position || Quaternion.Angle(camera.transform.rotation, rotation) > .0001f) error = "Camera moved during capture.";
            if (error != null || EditorApplication.timeSinceStartup - started > 120) { File.WriteAllText(root + "/status.txt", error ?? "timeout"); Dispose(); return; }
            if (step >= 32 && pending == 0) {
                if (++stage == Cases.Length) { File.WriteAllText(root + "/status.txt", "complete"); Dispose(); return; }
                StartStage();
            }
            EditorApplication.QueuePlayerLoopUpdate();
            UnityEditorInternal.InternalEditorUtility.RepaintAllViews();
        }
        private void Capture(ComputePassContext context, Texture depth, Texture normal, Texture shadow) {
            var c = context.Get<VividCameraData>();
            if (disposed || c.camera != camera || c.frameIndex == lastFrame || step >= 32) return;
            lastFrame = c.frameIndex; if (++warm < 64) return;
            var settings = VividVolumeManagerUtility.GetCascadedShadowSettingsVolume();
            var sd = context.Get<VividShadowData>(); var ld = context.Get<VividLightData>();
            var light = ld.mainLight.GetComponent<VividAdditionalLightData>();
            var frame = new Frame { width = shadow.width, height = shadow.height, frame = c.frameIndex, step = step,
                viewProjection = c.GetGPUViewProjectionMatrix(true), cameraPosition = camera.transform.position,
                cameraEuler = camera.transform.eulerAngles, lightDirection = -ld.mainLight.transform.forward,
                receiver = new Vector4(settings.virtualShadowMapPCF.value ? 1 : 0, sd.depthBias, sd.slopeScaleDepthBias, settings.virtualShadowMapStochasticFiltering.value ? 1 : 0),
                smrt = VirtualShadowMapReceiverQuality.BuildSMRTParameters(settings, light.angularDiameter),
                quality = VirtualShadowMapReceiverQuality.BuildParameters(settings),
                sampleIndexOffset = VirtualShadowMapReceiverQuality.BuildSMRTSampleIndexOffset(settings, additional, c.frameIndex),
                pages = VirtualShadowMapPrototypeRuntime.PhysicalPageCapacity, levels = sd.clipmaps.Count,
                state = VirtualShadowMapPrototypeRuntime.FrameState.ToString(), fallback = VirtualShadowMapPrototypeRuntime.LastFallbackReason.ToString(),
                aa = context.Get<VividAntialiasingData>().effectiveMode.ToString() };
            string prefix = directory + "/frame_" + step.ToString("D3");
            File.WriteAllText(prefix + ".json", JsonUtility.ToJson(frame));
            var cmd = context.cmd.m_WrappedCommandBuffer;
            if(!built) {cmd.BuildRayTracingAccelerationStructure(acceleration);built=true;}
            if(backDepth==null) {
                var pool=VirtualShadowMapPrototypeRuntime.StaticPhysicalPage.rt;
                backDepth=new RenderTexture(new RenderTextureDescriptor(pool.width,pool.height,GraphicsFormat.R32G32B32A32_UInt,0){enableRandomWrite=true,dimension=TextureDimension.Tex2DArray,volumeDepth=4});backDepth.Create();
            }
            if(step==0) {
                int build=audit.FindKernel("BuildAllDepth");
                cmd.SetRayTracingAccelerationStructure(audit,build,"_ResidualAcceleration",acceleration);
                cmd.SetComputeTextureParam(audit,build,"_CandidateDepthRW",backDepth);
                cmd.SetComputeBufferParam(audit,build,"_VSMPrototypePhysicalPageOwners",VirtualShadowMapPrototypeRuntime.PhysicalPageOwners);
                cmd.SetComputeBufferParam(audit,build,"_VSMProjections",VirtualShadowMapPrototypeRuntime.Projections.Buffer);
                cmd.SetComputeIntParam(audit,"_VSMPrototypePhysicalPageCapacity",frame.pages);
                cmd.SetComputeIntParam(audit,"_VSMPrototypePageSize",VirtualShadowMapPrototypeRuntime.PageSize);
                cmd.SetComputeIntParam(audit,"_VSMPrototypePagesPerAxis",VirtualShadowMapPrototypeRuntime.PagesPerAxis);
                cmd.SetComputeIntParam(audit,"_VSMPrototypePhysicalPagesPerRow",VirtualShadowMapPrototypeRuntime.PhysicalPagesPerRow);
                cmd.SetComputeIntParam(audit,"_VSMPrototypeVirtualResolution",sd.clipmaps.Resolution);
                cmd.DispatchCompute(audit,build,16,16,frame.pages);
            }
            if (step == 0) {
                File.WriteAllText(directory + "/settings.json", JsonUtility.ToJson(settings));
                if (readbackDepth == null) {
                    readbackDepth = new RenderTexture(new RenderTextureDescriptor(frame.width, frame.height, GraphicsFormat.R32_SFloat, 0) { enableRandomWrite = true });
                    readbackNormal = new RenderTexture(new RenderTextureDescriptor(frame.width, frame.height, GraphicsFormat.R32G32B32A32_SFloat, 0) { enableRandomWrite = true });
                    readbackDepth.Create(); readbackNormal.Create();
                }
                int copy = audit.FindKernel("VSMCopyReceiverData");
                cmd.SetComputeIntParam(audit, "_CSMOutputWidth", frame.width); cmd.SetComputeIntParam(audit, "_CSMOutputHeight", frame.height);
                cmd.SetComputeTextureParam(audit, copy, "_DepthTexture", depth); cmd.SetComputeTextureParam(audit, copy, "_GBuffer1", normal);
                cmd.SetComputeTextureParam(audit, copy, "_VSMCaptureDepth", readbackDepth); cmd.SetComputeTextureParam(audit, copy, "_VSMCaptureNormal", readbackNormal);
                cmd.DispatchCompute(audit, copy, (frame.width + 7) / 8, (frame.height + 7) / 8, 1);
                Save(cmd, readbackDepth, TextureFormat.RFloat, prefix + "_depth.bin.gz"); Save(cmd, readbackNormal, TextureFormat.RGBAFloat, prefix + "_normal.bin.gz");
                Save(cmd, VirtualShadowMapPrototypeRuntime.AllocatorCounters, prefix + "_counters.bin.gz");
                Save(cmd, VirtualShadowMapPrototypeRuntime.PageMetadata, prefix + "_metadata.bin.gz");
                Save(cmd, VirtualShadowMapPrototypeRuntime.PageTable, prefix + "_pagetable.bin.gz");
                Save(cmd, VirtualShadowMapPrototypeRuntime.Projections.Buffer, prefix + "_projections.bin.gz");
                
                additional.RequestFinalFrameScreenshot(directory + "/first.png");
            }
            DebugCapture(cmd, depth, normal, shadow, sd, frame, prefix);
            step++;
        }
        private void DebugCapture(CommandBuffer cmd, Texture depth, Texture normal, Texture shadow, VividShadowData sd, Frame f, string prefix) {
            if (data == null) {
                var desc = new RenderTextureDescriptor(f.width / 4, f.height / 4, GraphicsFormat.R32G32B32A32_SFloat, 0) { enableRandomWrite = true };
                data = new RenderTexture(desc); output = new RenderTexture(desc); data.Create(); output.Create();
            }
            cmd.SetComputeTextureParam(audit, kernel, "_DepthTexture", depth); cmd.SetComputeTextureParam(audit, kernel, "_GBuffer1", normal);
            cmd.SetComputeTextureParam(audit, kernel, "_VSMReceiverDebugShadow", shadow); cmd.SetComputeTextureParam(audit, kernel, "_VSMReceiverDebugData", data); cmd.SetComputeTextureParam(audit, kernel, "_VSMReceiverDebugOutput", output);
            cmd.SetComputeTextureParam(audit, kernel, "_VSMPrototypeStaticPhysicalPage", VirtualShadowMapPrototypeRuntime.StaticPhysicalPage); cmd.SetComputeTextureParam(audit, kernel, "_VSMPrototypeDynamicPhysicalPage", VirtualShadowMapPrototypeRuntime.DynamicPhysicalPage);
            cmd.SetComputeBufferParam(audit, kernel, "_VSMPrototypePageTable", VirtualShadowMapPrototypeRuntime.PageTable); cmd.SetComputeBufferParam(audit, kernel, "_VSMPrototypePageMetadata", VirtualShadowMapPrototypeRuntime.PageMetadata); cmd.SetComputeBufferParam(audit, kernel, "_VSMProjections", VirtualShadowMapPrototypeRuntime.Projections.Buffer);
            cmd.SetComputeIntParam(audit, "_VSMProjectionCount", f.levels); cmd.SetComputeIntParam(audit, "_VSMPrototypeEnabled", 1);
            cmd.SetComputeIntParam(audit, "_VSMPrototypeVirtualResolution", sd.clipmaps.Resolution); cmd.SetComputeIntParam(audit, "_VSMPrototypePageSize", VirtualShadowMapPrototypeRuntime.PageSize); cmd.SetComputeIntParam(audit, "_VSMPrototypePagesPerAxis", VirtualShadowMapPrototypeRuntime.PagesPerAxis); cmd.SetComputeIntParam(audit, "_VSMPrototypePhysicalPagesPerRow", VirtualShadowMapPrototypeRuntime.PhysicalPagesPerRow);
            cmd.SetComputeIntParam(audit, "_CSMOutputWidth", f.width); cmd.SetComputeIntParam(audit, "_CSMOutputHeight", f.height); cmd.SetComputeIntParam(audit, "_CSMFrameIndex", f.frame); cmd.SetComputeIntParam(audit, "_VSMSMRTSampleIndexOffset", f.sampleIndexOffset);
            cmd.SetComputeMatrixParam(audit, "_VSMReceiverViewProjection", f.viewProjection); cmd.SetComputeMatrixParam(audit, "_CSMInvViewProjMatrix", f.viewProjection.inverse);
            cmd.SetComputeVectorParam(audit, "_VSMReceiverParameters", f.receiver); cmd.SetComputeVectorParam(audit, "_VSMReceiverQuality", f.quality); cmd.SetComputeVectorParam(audit, "_VSMSMRTParameters", f.smrt);
            BlueNoise.Instance?.Bind(cmd, audit, kernel);
            cmd.SetRayTracingAccelerationStructure(audit,kernel,"_ResidualAcceleration",acceleration);
            cmd.SetComputeIntParam(audit,"_CandidateCount",stage==0?8:stage==1?10:stage==2?12:16);
            cmd.SetComputeTextureParam(audit,kernel,"_CandidateDepth",backDepth);
            foreach (int mode in new[]{5}) { cmd.SetComputeIntParam(audit, "_VSMReceiverDebugMode", mode); cmd.DispatchCompute(audit, kernel, (f.width / 4 + 7) / 8, (f.height / 4 + 7) / 8, 1); Save(cmd, data, TextureFormat.RGBAFloat, prefix + "_debug" + mode + ".bin.gz");  }
        }
        private void Save(CommandBuffer cmd, Texture texture, TextureFormat format, string path) { pending++; cmd.RequestAsyncReadback(texture, 0, format, r => Complete(r, path)); }
        private void Save(CommandBuffer cmd, GraphicsBuffer buffer, string path) { pending++; cmd.RequestAsyncReadback(buffer, r => Complete(r, path)); }
        private void Complete(AsyncGPUReadbackRequest request, string path) {
            try { if (request.hasError) error = "Readback failed: " + path;
                else { using var file = File.Create(path); using var zip = new GZipStream(file, System.IO.Compression.CompressionLevel.Fastest); byte[] bytes = request.GetData<byte>().ToArray(); zip.Write(bytes, 0, bytes.Length); }
            } catch (Exception e) { error = e.ToString(); } finally { pending--; }
        }
        public void Dispose() {
            if (disposed) return; disposed = true;
            EditorApplication.update -= Tick; AssemblyReloadEvents.beforeAssemblyReload -= Dispose; CSMShadowResolvePass.EditorReceiverCapture -= Capture;
            AsyncGPUReadback.WaitAllRequests();
            Object.DestroyImmediate(volumeObject); Object.DestroyImmediate(profile); Object.DestroyImmediate(audit); Object.DestroyImmediate(data); Object.DestroyImmediate(output); Object.DestroyImmediate(readbackDepth); Object.DestroyImmediate(readbackNormal);
            Object.DestroyImmediate(backDepth); acceleration?.Dispose(); rays?.Dispose();
            Application.runInBackground = background; active = null;
            EditorApplication.QueuePlayerLoopUpdate();
        }
    }
}
