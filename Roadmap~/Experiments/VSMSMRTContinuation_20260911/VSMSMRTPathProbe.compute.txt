#pragma kernel UploadGeometryDepth
#pragma kernel CompareGeometryRays VIVID_VSM_RECEIVER_DEBUG
#include "../../Shaders/Core/Private/CSMShadowResolve.compute"
StructuredBuffer<float> _GeometryDepth;
RWTexture2D<uint> _GeometryStatic, _GeometryDynamic;
StructuredBuffer<float4> _GeometryOrigins, _GeometryRays;
RWStructuredBuffer<float4> _GeometryResults;
int _GeometryCount, _GeometryBudget;
[numthreads(8,8,1)]
void UploadGeometryDepth(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_VSMPrototypeVirtualResolution || id.y >= (uint)(_VSMPrototypeVirtualResolution * _VSMProjectionCount)) return;
    _GeometryStatic[id.xy] = asuint(_GeometryDepth[id.y * _VSMPrototypeVirtualResolution + id.x]);
    _GeometryDynamic[id.xy] = 0;
}
#if defined(VIVID_VSM_RECEIVER_DEBUG)
[numthreads(64,1,1)]
void CompareGeometryRays(uint3 id : SV_DispatchThreadID)
{
    if (id.x >= (uint)_GeometryCount) return;
    float3 origin = _GeometryOrigins[id.x].xyz;
    g_VSMDebugWork = 0;
    float visibility;
    bool valid = TryTraceVSMSMRTClipmaps(origin, _GeometryRays[id.x].xy, .01, _GeometryBudget, 0, visibility);
    uint work = g_VSMDebugWork.x;
    bool support = HasVSMSMRTFootprint(origin.xy, 0);
    _GeometryResults[id.x] = float4(valid ? 1 : 0, visibility, work, support ? 1 : 0);
}

#endif
