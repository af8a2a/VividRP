using System;
using System.IO;
using Unity.Mathematics;
using UnityEditor;
using UnityEngine;
using UnityEngine.Experimental.Rendering;
using Object = UnityEngine.Object;

namespace VividRP.Editor
{
    // Explicit, synchronous diagnostic. Never used by the render loop or NUnit.
    internal static class VSMSMRTPathProbe
    {
        [MenuItem("Tools/VividRP/Diagnostics/Run SMRT Path Reference")]
        private static void Run()
        {
            if (EditorApplication.isPlaying) throw new InvalidOperationException("Run this isolated probe in Edit mode.");
            const string path = "Packages/com.vivid.render-pipelines/Editor/Tools/VSMSMRTPathProbe.compute";
            string root = UnityEditor.PackageManager.PackageInfo.FindForAssetPath(path).resolvedPath;
            string work = Path.Combine(root, "Temp~/smrt-path");
            int completed = 0;
            string error = null;
            ComputeShader shader = null;
            try
            {
                shader = Object.Instantiate(AssetDatabase.LoadAssetAtPath<ComputeShader>(path));
                using var reader = new BinaryReader(File.OpenRead(Path.Combine(work, "inputs.bin")));
                using var writer = new BinaryWriter(File.Create(Path.Combine(work, "gpu.bin")));
                int cases = reader.ReadInt32(); writer.Write(cases);
                for (int i = 0; i < cases; i++)
                {
                    Compare(shader, reader, writer);
                    completed++;
                }
                if (reader.BaseStream.Position != reader.BaseStream.Length) throw new InvalidDataException("Trailing input data");
            }
            catch (Exception e) { error = e.ToString(); }
            finally
            {
                if (shader != null) Object.DestroyImmediate(shader);
                string status = "gpu=" + SystemInfo.graphicsDeviceType + " device=" + SystemInfo.graphicsDeviceName
                    + " unity=" + Application.unityVersion + " completed=" + completed + " error=" + (error ?? "none");
                File.WriteAllText(Path.Combine(work, "gpu-status.txt"), status);
                if (error != null) Debug.LogError(status); else Debug.Log(status);
            }
        }

        private static RenderTexture Pool(int size, int levels)
        {
            var texture = new RenderTexture(new RenderTextureDescriptor(size, size * levels)
            {
                graphicsFormat = GraphicsFormat.R32_UInt, depthStencilFormat = GraphicsFormat.None,
                enableRandomWrite = true, msaaSamples = 1,
            });
            if (!texture.Create())
            {
                Object.DestroyImmediate(texture);
                throw new InvalidOperationException("Cannot create depth pool");
            }
            return texture;
        }

        private static float4[] ReadVectors(BinaryReader reader, int count)
        {
            var values = new float4[count];
            for (int i = 0; i < count; i++) values[i] = new float4(reader.ReadSingle(), reader.ReadSingle(), reader.ReadSingle(), reader.ReadSingle());
            return values;
        }

        private static void Compare(ComputeShader shader, BinaryReader reader, BinaryWriter writer)
        {
            int size = reader.ReadInt32(), count = reader.ReadInt32(), budget = reader.ReadInt32(), levels = reader.ReadInt32();
            float maxLength = reader.ReadSingle(), tangent = reader.ReadSingle();
            var projections = new VividRP.Runtime.RenderPass.Core.VirtualShadowMapProjection[levels];
            var tableData = new uint[levels * 4]; var metadataData = new uint4[levels * 4];
            for (int level = 0; level < levels; level++)
            {
                float w = reader.ReadSingle(); reader.ReadSingle();
                float cx = reader.ReadSingle(), cy = reader.ReadSingle(), zs = reader.ReadSingle(), zo = reader.ReadSingle();
                var matrix = Matrix4x4.identity;
                matrix.m00 = matrix.m11 = 1 / (w * size); matrix.m22 = zs;
                matrix.m03 = .5f - cx / (w * size); matrix.m13 = .5f - cy / (w * size); matrix.m23 = zo;
                projections[level].WorldToShadow = matrix; projections[level].Parameters = new Vector4(w, 0, 0, 100);
                for (int page = level * 4; page < level * 4 + 4; page++)
                {
                    uint encoded = (uint)(levels * 4 - page);
                    tableData[page] = encoded; metadataData[page] = new uint4(10, encoded, 7, 8);
                }
            }
            var depth = new float[size * size * levels];
            for (int i = 0; i < depth.Length; i++) depth[i] = reader.ReadSingle();
            // Reverse all physical slots; crossing a virtual edge cannot rely on
            // physically adjacent pages, even within the same clipmap segment.
            var physicalDepth = new float[depth.Length];
            int pageSize = size / 2;
            for (int level = 0; level < levels; level++) for (int y = 0; y < size; y++) for (int x = 0; x < size; x++)
            {
                int page = level * 4 + y / pageSize * 2 + x / pageSize;
                int slot = (int)tableData[page] - 1;
                physicalDepth[(slot / 2 * pageSize + y % pageSize) * size + slot % 2 * pageSize + x % pageSize]
                    = depth[(level * size + y) * size + x];
            }
            depth = physicalDepth;
            var origins = ReadVectors(reader, count);
            var rays = ReadVectors(reader, count);
            using var upload = new GraphicsBuffer(GraphicsBuffer.Target.Structured, depth.Length, 4);
            using var input = new GraphicsBuffer(GraphicsBuffer.Target.Structured, count, 16);
            using var directions = new GraphicsBuffer(GraphicsBuffer.Target.Structured, count, 16);
            using var output = new GraphicsBuffer(GraphicsBuffer.Target.Structured, count, 16);
            using var table = new GraphicsBuffer(GraphicsBuffer.Target.Structured, levels * 4, 4);
            using var metadata = new GraphicsBuffer(GraphicsBuffer.Target.Structured, levels * 4, 16);
            using var projectionBuffer = new GraphicsBuffer(GraphicsBuffer.Target.Structured, levels, 160);
            projectionBuffer.SetData(projections);
            RenderTexture staticPool = null, dynamicPool = null;
            try
            {
                staticPool = Pool(size, levels); dynamicPool = Pool(size, levels);
                upload.SetData(depth); input.SetData(origins); directions.SetData(rays);
                table.SetData(tableData); metadata.SetData(metadataData);
                shader.SetInt("_VSMPrototypeEnabled", 1); shader.SetInt("_VSMProjectionCount", levels);
                shader.SetVector("_VSMSMRTParameters", new Vector4(4, budget, maxLength, tangent));
                shader.SetInt("_VSMPrototypeVirtualResolution", size); shader.SetInt("_VSMPrototypePageSize", pageSize);
                shader.SetInt("_VSMPrototypePagesPerAxis", 2); shader.SetInt("_VSMPrototypePhysicalPagesPerRow", 2);
                shader.SetInt("_VSMPrototypePageTableEntryCount", levels * 4); shader.SetInt("_VSMPrototypePhysicalPageCapacity", levels * 4);
                int kernel = shader.FindKernel("UploadGeometryDepth");
                shader.SetBuffer(kernel, "_GeometryDepth", upload);
                shader.SetTexture(kernel, "_GeometryStatic", staticPool); shader.SetTexture(kernel, "_GeometryDynamic", dynamicPool);
                shader.Dispatch(kernel, (size + 7) / 8, (size * levels + 7) / 8, 1);
                kernel = shader.FindKernel("CompareGeometryRays");
                shader.SetBuffer(kernel, "_VSMProjections", projectionBuffer);
                shader.SetInt("_GeometryCount", count); shader.SetInt("_GeometryBudget", budget);
                shader.SetBuffer(kernel, "_GeometryOrigins", input); shader.SetBuffer(kernel, "_GeometryRays", directions);
                shader.SetBuffer(kernel, "_GeometryResults", output);
                shader.SetBuffer(kernel, "_VSMPrototypePageTable", table); shader.SetBuffer(kernel, "_VSMPrototypePageMetadata", metadata);
                shader.SetTexture(kernel, "_VSMPrototypeStaticPhysicalPage", staticPool);
                shader.SetTexture(kernel, "_VSMPrototypeDynamicPhysicalPage", dynamicPool);
                shader.Dispatch(kernel, (count + 63) / 64, 1, 1);
                var values = new float4[count]; output.GetData(values);
                for (int i = 0; i < values.Length; i++) { writer.Write(values[i].x); writer.Write(values[i].y); writer.Write(values[i].z); writer.Write(values[i].w); }
            }
            finally
            {
                if (staticPool != null) Object.DestroyImmediate(staticPool);
                if (dynamicPool != null) Object.DestroyImmediate(dynamicPool);
            }
        }
    }
}
