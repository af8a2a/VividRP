using System;
using System.IO;
using System.Reflection;
using UnityEditor;
using UnityEngine;
using VividRP.Runtime;
using VividRP.Runtime.RenderPass.Core;
namespace VividRP.Editor {
internal static class VSMSMRTTiming {
 static VSMBaselineSession session;
 static double started;
 static int stage, originalLimit;
 static string root;
 static VividAdditionalLightData light;
 static float originalAngle;
 static readonly string[] Labels={"pcf512_a","smrt8x4_a","smrt8x8_a","smrt4x8_a","smrt4x8_b","smrt8x8_b","smrt8x4_b","pcf512_b"};
 static readonly int[] Limits={512,512,512,512,512,512,512,512};
 [MenuItem("Tools/VividRP/Diagnostics/Run Temporary VSM SMRT Timing")]
 static void Run(){
  if(Application.dataPath!="E:/VividRP_Reborn/Assets"||!EditorApplication.isPlaying||session!=null)throw new InvalidOperationException("Use project Play Mode.");
  originalLimit=VirtualShadowMapPrototypeRuntime.EditorPhysicalPageLimit;
  foreach(var l in UnityEngine.Object.FindObjectsByType<Light>(FindObjectsSortMode.None))if(l.type==LightType.Directional){light=l.GetComponent<VividAdditionalLightData>();break;}
  if(light==null)throw new InvalidOperationException("No sun");originalAngle=light.angularDiameter;light.angularDiameter=.5f;
  root="Packages/VividRP/Temp~/vsm-smrt-timing/"+DateTime.UtcNow.ToString("yyyyMMdd_HHmmss_fff");Directory.CreateDirectory(root);stage=0;
  try{Begin();}catch{Finish();throw;}
  EditorApplication.update+=Tick;AssemblyReloadEvents.beforeAssemblyReload+=Finish;
 }
 static void Begin(){
  VirtualShadowMapPrototypeRuntime.EditorPhysicalPageLimit=Limits[stage];VirtualShadowMapPrototypeRuntime.ReleaseResources();
  var c=new VSMBaselineCase{name=Labels[stage],resolution=4096,firstLevel=0,screenDensity=true,pcf=true,transition=.2f,maxDistance=150,screenSpaceDenoise=true};
  session=new VSMBaselineSession(Camera.main,new[]{c},5,5,120,4096,root+"/"+Labels[stage],"4096 SMRT fixed layout and matched capacity",false);
  // Cold diagnostic setup only. Keep shipping timing classes unchanged.
  var settings=(CascadedShadowSettingsVolume)typeof(VSMBaselineSession).GetField("m_Settings",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(session);
  settings.virtualShadowMapSMRT.Override(!Labels[stage].StartsWith("pcf",StringComparison.Ordinal));
  settings.virtualShadowMapSMRTRayCount.Override(Labels[stage].StartsWith("smrt8",StringComparison.Ordinal)?8:4);
  settings.virtualShadowMapSMRTSamplesPerRay.Override(Labels[stage].StartsWith("smrt4x4",StringComparison.Ordinal)||Labels[stage].StartsWith("smrt8x4",StringComparison.Ordinal)?4:8);
  settings.virtualShadowMapSMRTMaxRayLength.Override(10);
  File.WriteAllText(root+"/"+Labels[stage]+"/smrt-settings.json",JsonUtility.ToJson(settings,true));
  started=EditorApplication.timeSinceStartup;
 }
 static void Tick(){
  if(session==null)return;
  if(session.Tick()){
   session.Finish("completed");session.Dispose();session=null;
   if(++stage<Labels.Length){Begin();return;}
   File.WriteAllText(root+"/status.txt","complete");Finish();return;
  }
  if(session.IsFinished||EditorApplication.timeSinceStartup-started>60){File.WriteAllText(root+"/status.txt","incomplete stage="+stage);Finish();}
 }
 static void Finish(){
  EditorApplication.update-=Tick;AssemblyReloadEvents.beforeAssemblyReload-=Finish;
  if(session!=null){session.Dispose();session=null;}VirtualShadowMapPrototypeRuntime.EditorPhysicalPageLimit=originalLimit;
  if(light!=null)light.angularDiameter=originalAngle;
  if(EditorApplication.isPlaying)EditorApplication.isPlaying=false;
 }
}}
