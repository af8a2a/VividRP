#define VIVID_MAX_MESHLET_TRIANGLES 128u
#define VIVID_VISIBILITY_INSTANCE_ID_BIAS 1u
#define VIVID_VISIBILITY_TRIANGLE_INDEX_BITS 8u
#define VIVID_VISIBILITY_TRIANGLE_INDEX_MASK 255u
#define VIVID_MESHLET_NORMAL_COMPONENT_MASK 0x7FFFu

float3 VividSafeNormalize(const float3 value)
{
    return value * rsqrt(max(dot(value, value), 1e-20f));
}

float3 DecodeVividMeshletOctahedral15(const uint packedDirection)
{
    float2 octahedral = float2(
        packedDirection & VIVID_MESHLET_NORMAL_COMPONENT_MASK,
        (packedDirection >> 15u) & VIVID_MESHLET_NORMAL_COMPONENT_MASK);
    octahedral = octahedral / float(VIVID_MESHLET_NORMAL_COMPONENT_MASK) * 2.0f - 1.0f;

    float3 direction = float3(
        octahedral,
        1.0f - abs(octahedral.x) - abs(octahedral.y));
    const float fold = saturate(-direction.z);
    const float2 foldSign = step(0.0f, direction.xy) * 2.0f - 1.0f;
    direction.xy -= foldSign * fold;
    return VividSafeNormalize(direction);
}

float3 DecodeVividMeshletNormal(const uint packedNormal)
{
    return (packedNormal & VIVID_MESHLET_NORMAL_VALID_BIT) != 0u
        ? DecodeVividMeshletOctahedral15(packedNormal)
        : 0.0f;
}

uint PullMeshletIndex(const VividMeshlet meshlet, const uint indexID)
{
    const uint absoluteIndexID = meshlet.TriangleOffset + indexID;
    const uint packedIndices = _SharedIndexBuffer.Load((absoluteIndexID / 4u) * 4u);
    return (packedIndices >> ((absoluteIndexID % 4u) * 8u)) & 0xFFu;
}

uint2 PackVisibilityValue(
    const VividMeshletRenderRequestPacked renderRequest,
    const uint triangleIndex)
{
    return uint2(
        renderRequest.InstanceID_LOD + VIVID_VISIBILITY_INSTANCE_ID_BIAS,
        (renderRequest.MeshletID << VIVID_VISIBILITY_TRIANGLE_INDEX_BITS)
            | (triangleIndex & VIVID_VISIBILITY_TRIANGLE_INDEX_MASK));
}

float2 EncodeVisibilityNormalOct(const float3 normalWS)
{
    float3 normal = VividSafeNormalize(normalWS);
    normal /= max(abs(normal.x) + abs(normal.y) + abs(normal.z), 1e-6f);
    float2 encoded = normal.xy;
    if (normal.z < 0.0f)
    {
        const float2 signs = float2(
            encoded.x >= 0.0f ? 1.0f : -1.0f,
            encoded.y >= 0.0f ? 1.0f : -1.0f);
        encoded = (1.0f - abs(encoded.yx)) * signs;
    }
    return encoded * 0.5f + 0.5f;
}

[numthreads(1, 1, 1)]
void AmplificationMain()
{
    const uint argsAddress = _RendererListIndex * VIVID_INDIRECT_DRAW_ARGS_STRIDE;
    const uint startInstance = _VisibleMeshletIndirectArgs.Load(
        argsAddress + VIVID_INDIRECT_DRAW_ARGS_START_INSTANCE_OFFSET);
    const uint availableRequestCount = startInstance < _MaxRequestCount
        ? _MaxRequestCount - startInstance
        : 0u;
    const uint instanceCount = min(
        _VisibleMeshletIndirectArgs.Load(
            argsAddress + VIVID_INDIRECT_DRAW_ARGS_INSTANCE_COUNT_OFFSET),
        min(availableRequestCount, VIVID_DISPATCH_MESH_MAX_TOTAL_GROUPS));
    const uint dispatchGroupCountX = min(
        instanceCount,
        VIVID_DISPATCH_MESH_MAX_GROUPS_PER_DIMENSION);
    const uint dispatchGroupCountY = dispatchGroupCountX == 0u
        ? 1u
        : (instanceCount + dispatchGroupCountX - 1u) / dispatchGroupCountX;

    g_MeshDispatchPayload.StartInstance = startInstance;
    g_MeshDispatchPayload.InstanceCount = instanceCount;
    g_MeshDispatchPayload.DispatchGroupCountX = max(dispatchGroupCountX, 1u);
    DispatchMesh(dispatchGroupCountX, dispatchGroupCountY, 1u, g_MeshDispatchPayload);
}

[numthreads(128, 1, 1)]
[outputtopology("triangle")]
void MeshMain(
    in payload VividMeshDispatchPayload meshPayload,
    const uint3 groupID : SV_GroupID,
    const uint groupThreadID : SV_GroupThreadID,
    out vertices VividMeshVertexOutput outputVertices[VIVID_MAX_MESHLET_VERTICES],
    out indices uint3 outputTriangles[VIVID_MAX_MESHLET_TRIANGLES],
    out primitives VividMeshPrimitiveOutput outputPrimitives[VIVID_MAX_MESHLET_TRIANGLES])
{
    const uint requestOffset = groupID.x + groupID.y * meshPayload.DispatchGroupCountX;
    const bool requestIsValid = requestOffset < meshPayload.InstanceCount;
    const uint safeRequestOffset = requestIsValid ? requestOffset : 0u;
    const VividMeshletRenderRequestPacked renderRequest =
        _VisibleMeshletRenderRequests[meshPayload.StartInstance + safeRequestOffset];
    const VividInstanceData instanceData = _InstanceData[renderRequest.InstanceID_LOD];
    const VividMeshlet meshlet = _Meshlets[renderRequest.MeshletID];
    const uint vertexCount = requestIsValid
        ? meshlet.PackedVertexTriangleCounts & 0xFFFFu
        : 0u;
    const uint triangleCount = requestIsValid
        ? meshlet.PackedVertexTriangleCounts >> 16u
        : 0u;

    SetMeshOutputCounts(vertexCount, triangleCount);

    if (!requestIsValid)
        return;

    if (groupThreadID < vertexCount)
    {
        const VividMeshletVertex vertex =
            _SharedVertexBuffer[meshlet.VertexOffset + groupThreadID];
        const float3 positionOS = float3(vertex.PositionX, vertex.PositionY, vertex.PositionZ);
        const float3 positionWS = mul(instanceData.ObjectToWorldMatrix, float4(positionOS, 1.0f)).xyz;

        VividMeshVertexOutput output;
        output.positionCS = mul(_ViewProjectionMatrix, float4(positionWS, 1.0f));
        output.uv0 = vertex.UV;
        output.geometricNormalWS = VividSafeNormalize(
            mul(
                DecodeVividMeshletNormal(vertex.PackedNormal),
                (float3x3)instanceData.WorldToObjectMatrix));
        outputVertices[groupThreadID] = output;
    }

    if (groupThreadID < triangleCount)
    {
        const uint firstIndex = groupThreadID * 3u;
        outputTriangles[groupThreadID] = uint3(
            PullMeshletIndex(meshlet, firstIndex),
            PullMeshletIndex(meshlet, firstIndex + 1u),
            PullMeshletIndex(meshlet, firstIndex + 2u));
        outputPrimitives[groupThreadID].visibilityValue =
            PackVisibilityValue(renderRequest, groupThreadID);
    }
}

VividVisibilityBufferFragmentOutput PixelMain(
    VividMeshVertexOutput input,
    nointerpolation uint2 visibilityValue : TEXCOORD2,
    linear float3 barycentrics : SV_Barycentrics)
{
    const float2 uv0Ddx = ddx(input.uv0);
    const float2 uv0Ddy = ddy(input.uv0);

    VividVisibilityBufferFragmentOutput output;
    output.visibility = visibilityValue;
    output.attributes0 = float4(input.uv0, uv0Ddx);
    output.attributes1 = float4(uv0Ddy, EncodeVisibilityNormalOct(input.geometricNormalWS));
    output.barycentrics = barycentrics.xy;
    return output;
}
