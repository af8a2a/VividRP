    void WriteOracleFixture(ComputePassContext context) {
        var sceneData=PrivateField("m_Scene");var read=sceneData.GetType().GetMethod("ReadDiffuseSurface",inst);
        float[] V(Vector3 v)=>new[]{v.x,v.y,v.z};
        var geometry=roots.SelectMany(r=>r.GetComponentsInChildren<VividRP.Runtime.GPUDriven.MeshletRenderer>())
            .Where(r=>r.isActiveAndEnabled && r.sourceRenderingEnabled).OrderBy(r=>r.name).Select(r=>{
                var surface=read.Invoke(sceneData,new object[]{r});var t=surface.GetType();
                var albedo=(Vector4)t.GetField("AlbedoAndValid").GetValue(surface);var emission=(Vector4)t.GetField("Emission").GetValue(surface);
                if(albedo.w<=0f || emission.w!=2f)throw new InvalidOperationException("Oracle requires valid constant Cull Back Lambert surfaces: "+r.name);
                return new{name=r.name,position=V(r.transform.position),rotation=V(r.transform.eulerAngles),scale=V(r.transform.lossyScale),
                    meshBounds=V(r.sourceMesh.bounds.size),meshCenter=V(r.sourceMesh.bounds.center),baseColorLinear=new[]{albedo.x,albedo.y,albedo.z},
                    emissionLinear=new[]{emission.x,emission.y,emission.z},cullMode=(int)emission.w,sourceMaterial=r.sourceMaterials[0].name,
                    shader=r.sourceMaterials[0].shader.name};
            }).ToArray();
        if(geometry.Length!=8 || context.Get<VividSkyData>().activeSkyType!=SkyType.None)throw new InvalidOperationException("Oracle fixture/environment changed.");
        var fixture=new{source=scene.path,geometry,camera=new{position=V(camera.transform.position),euler=V(camera.transform.eulerAngles),
            verticalFov=camera.fieldOfView,aspect=1920.0/1080.0},quantity="Incident Lambert diffuse E/pi, scene linear",frame,
            note="Effective ReadDiffuseSurface values from the current isolated scene; primary receiver normals exported separately. No source assets saved."};
        string path=output+"/live-fixture.json";
        if(System.IO.File.Exists(path))throw new InvalidOperationException("Refusing to overwrite oracle fixture.");
        System.IO.File.WriteAllText(path,Newtonsoft.Json.JsonConvert.SerializeObject(fixture,Newtonsoft.Json.Formatting.Indented));oracleFixtureWritten=true;
        records.Add(new{oracleFixturePath=path,fixtureSha256=Hash(System.IO.File.ReadAllBytes(path))});
    }
    void ReadOraclePixels(ComputePassContext context,Texture texture,string role,string side,bool fullPixel=false) {
        int f=frame;string format=texture.graphicsFormat.ToString();int stride=format=="R16G16B16A16_SFloat"?8:format=="R32G32B32A32_SFloat"?16:0;
        if(stride==0 || (fullPixel && (texture.width!=1920 || texture.height!=1080)))throw new InvalidOperationException("Unexpected oracle source descriptor.");
        pending++;
        ((CommandBuffer)wrapped.GetValue(context.cmd)).RequestAsyncReadback(texture,0,request=>{
            try {
                if(request.hasError || Environment.CurrentManagedThreadId!=callbackThread)throw new InvalidOperationException("Oracle readback failed.");
                var bytes=request.GetData<byte>().ToArray();var crop=new byte[oracleCodes.Length*stride];
                for(int i=0;i<oracleCodes.Length;++i)Buffer.BlockCopy(bytes,fullPixel?(int)oracleCodes[i]*stride:i*stride,crop,i*stride,stride);
                string path=output+"/"+arm+"/"+f.ToString("0000")+"-oracle-"+role+"-"+side+".bin";Write(path,crop);
                records.Add(new{frame=f,tag="oracle-"+role,side,width=oracleCodes.Length,height=1,format,path,saved=true,sha256=Hash(crop)});
            }catch(Exception e){errors.Add(e.Message);}finally{pending--;}
        });
    }
