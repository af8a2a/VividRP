    void ReadRecoveryEligibility(ComputePassContext context,Texture texture) {
        int f=frame;pending++;
        ((CommandBuffer)wrapped.GetValue(context.cmd)).RequestAsyncReadback(texture,0,request=>{
            try {
                if(request.hasError || Environment.CurrentManagedThreadId!=callbackThread)throw new InvalidOperationException("Recovery eligibility readback failed.");
                var values=request.GetData<ushort>();
                foreach(var roi in privateRois) {
                    int eligible=0;
                    for(int y=0;y<roi.Item5;++y)for(int x=0;x<roi.Item4;++x) {
                        float a=Unity.Mathematics.math.f16tof32(values[((roi.Item3+y)*1920+roi.Item2+x)*4+3]);
                        if(!float.IsFinite(a) || Math.Abs(a)>1f)throw new InvalidOperationException("Invalid recovery coverage.");
                        if(a<0){if(a>-.4995f)throw new InvalidOperationException("Weak quadrature support.");eligible++;}
                    }
                    records.Add(new{frame=f,recoveryEligibility=roi.Item1,eligible,pixels=roi.Item4*roi.Item5});
                }
            }catch(Exception e){errors.Add(e.Message);}finally{pending--;}
        });
    }
