    var timingRecorders=new ProfilerRecorder[2];var timingSamples=new List<object>();

    void StartTiming() {
        if(timingStarted || pending!=0)throw new InvalidOperationException("Readbacks did not drain before timing.");
        UnityEditorInternal.ProfilerDriver.SetAreaEnabled(UnityEngine.Profiling.ProfilerArea.GPU,true);
        UnityEditorInternal.ProfilerDriver.profileEditor=false;UnityEngine.Profiling.Profiler.enabled=true;
        const ProfilerRecorderOptions options=ProfilerRecorderOptions.StartImmediately
            | ProfilerRecorderOptions.SumAllSamplesInFrame | ProfilerRecorderOptions.GpuRecorder;
        for(int i=0;i<2;++i)timingRecorders[i]=ProfilerRecorder.StartNew(ProfilerCategory.Render,timingNames[i],1024,options);
        timingStarted=true;
    }
    void FinishTiming() {
        if(!timingStarted)throw new InvalidOperationException("GPU timing never started.");
        for(int i=0;i<2;++i) {
            var values=new List<ProfilerRecorderSample>();timingRecorders[i].CopyTo(values);
            timingSamples.Add(new{marker=timingNames[i],valid=timingRecorders[i].Valid,count=timingRecorders[i].Count,
                wrappedAround=timingRecorders[i].WrappedAround,samples=values.Select(v=>new{nanoseconds=v.Value,count=v.Count}).ToArray()});
            timingRecorders[i].Stop();timingRecorders[i].Dispose();timingRecorders[i]=default;
        }
    }
    void CleanupTiming() {
        for(int i=0;i<2;++i)if(timingRecorders[i].Valid)timingRecorders[i].Dispose();
        UnityEngine.Profiling.Profiler.enabled=previousTimingProfiler;
        UnityEditorInternal.ProfilerDriver.profileEditor=previousTimingEditor;
        UnityEditorInternal.ProfilerDriver.SetAreaEnabled(UnityEngine.Profiling.ProfilerArea.GPU,previousTimingGpu);
    }
