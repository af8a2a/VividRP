float PackRadianceHistoryAge(float age, float stale)
{
    uint count = uint(round(clamp(age, 1.0, 512.0))) - 1u;
    uint undrawn = uint(round(clamp(stale, 0.0, 8.0)));
    return f16tof32(0x400u + count + (undrawn << 9u));
}

float2 UnpackRadianceHistoryAge(float value)
{
    uint bits = f32tof16(value);
    if (bits < 0x400u || bits >= 0x2400u)
        return 0.0;
    uint payload = bits - 0x400u;
    return float2(float(payload & 511u) + 1.0, float((payload >> 9u) & 15u));
}

[numthreads(16, 1, 1)]
void ScreenProbeTemporallyAccumulateTraceRadianceCS(uint3 group : SV_GroupID, uint bin : SV_GroupIndex)
{
    uint2 tile = group.xy;
    uint2 grid = DirectionalGrid();
    uint2 offset = DirectionalMetadataOffset(grid);
    uint2 address = DirectionalGatherAddress(tile, bin);
    float4 geometry = _SdfDiffuseGeometry[tile];
    float4 sample = _SdfDirectionalComposite.Load(int3(address, 0));
    float4 hit = _SdfDirectionalComposite.Load(int3(address + offset, 0));
    float4 plane = _SdfDirectionalComposite.Load(int3(address + 2u * offset, 0));
    // Composite stores conditional radiance and inverse-PDF covered mass.
    // Scatter that mass into an equal-cosine-mass radiance grid. Undrawn cells
    // contribute zero this frame; they are not omitted from the estimator.
    const float cellMass = float(RaysPerProbe) / float(DirectionalGatherBinCount);
    float3 radiance = sample.rgb * sample.a / cellMass;
    float age = 1.0;
    float stale = 0.0;
    if (geometry.w >= 0.0 && _SdfDirectionalHasHistory != 0u)
    {
        float4 clip = mul(_SdfDiffusePreviousVP, float4(geometry.xyz, 1.0));
        float2 uv = ScreenUv(clip);
        if (clip.w > 0.0 && all(uv >= 0.0) && all(uv < 1.0))
        {
            float3 normal = UnpackProbeNormal(geometry.w);
            float3 direction = DirectionalGatherDirection(bin, GetLocalFrame(normal));
            float2 probe = ProbeHistoryGridPosition(uv);
            int2 lower = int2(floor(probe));
            float2 fraction = frac(probe);
            float eyeDepth = max(0.1, dot(float4(geometry.xyz, 1.0), _SdfProbeCameraForward));
            float3 history = 0.0;
            float total = 0.0, ageSum = 0.0, bestWeight = 0.0;
            float4 historyHit = 0.0, historyPlane = 0.0;
            [unroll]
            for (uint tap = 0u; tap < 4u; ++tap)
            {
                int2 corner = int2(tap & 1u, tap >> 1u);
                int2 candidate = lower + corner;
                if (any(candidate < 0) || any(candidate >= int2(grid)))
                    continue;
                uint2 oldTile = uint2(candidate);
                float4 other = _SdfDiffusePreviousGeometry.Load(int3(oldTile, 0));
                float weight = ProbeTraceGeometryWeight(other, geometry.xyz, normal, eyeDepth);
                if (_SdfDiffuseAdaptiveProbes != 0u)
                {
                    uint2 alternateTile = oldTile + uint2(0u, grid.y);
                    float4 alternate = _SdfDiffusePreviousGeometry.Load(int3(alternateTile, 0));
                    float alternateWeight = ProbeTraceGeometryWeight(alternate, geometry.xyz, normal, eyeDepth);
                    if (alternateWeight > weight)
                    {
                        oldTile = alternateTile;
                        other = alternate;
                        weight = alternateWeight;
                    }
                }
                if (weight <= 0.0 || length(other.xyz - geometry.xyz) > clamp(0.035 * eyeDepth, 0.12, 0.40))
                    continue;
                uint oldBin = DirectionalGatherBin(direction, GetLocalFrame(UnpackProbeNormal(other.w)));
                uint2 oldAddress = DirectionalGatherAddress(oldTile, oldBin);
                float4 old = _SdfDirectionalPrevious.Load(int3(oldAddress, 0));
                if (old.a <= 0.0)
                    continue;
                float2 bilinear = lerp(1.0 - fraction, fraction, float2(corner));
                weight *= bilinear.x * bilinear.y;
                float4 oldHit = _SdfDirectionalPrevious.Load(int3(oldAddress + offset, 0));
                history += old.rgb * _SdfHistoryLightScale.rgb * weight;
                total += weight;
                ageSum += UnpackRadianceHistoryAge(oldHit.y).x * weight;
                if (weight > bestWeight)
                {
                    bestWeight = weight;
                    historyHit = oldHit;
                    historyPlane = _SdfDirectionalPrevious.Load(int3(oldAddress + 2u * offset, 0));
                }
            }
            if (total > 1e-5)
            {
                // Lumen's probe temporal filter blends radiance after scatter,
                // including zero-valued undrawn cells, using geometric reprojection.
                age = min(1.0 + ageSum / total, 512.0);
                // Ramp history support after reset. A long fixed blend must not
                // retain the first sparse frame throughout the warmup window.
                float blend = min(0.5, (age - 1.0) / age);
                radiance = lerp(radiance, history / total, blend);
                if (hit.x == 0.0 && hit.y == 0.0)
                {
                    stale = min(UnpackRadianceHistoryAge(historyHit.y).y + 1.0, 8.0);
                    if (stale < 8.0)
                    {
                        hit = historyHit;
                        plane = historyPlane;
                    }
                }
            }
        }
    }
    hit.y = PackRadianceHistoryAge(age, stale);
    // Alpha now validates the radiance-grid receiver, not per-ray coverage or
    // history confidence. Unresolved ray energy remains zero in RGB.
    _SdfDirectionalTemporalOutput[address] = RoundSdfHistory(geometry.w >= 0.0 ? float4(radiance, 1.0) : 0.0);
    _SdfDirectionalTemporalOutput[address + offset] = RoundSdfHistory(hit);
    _SdfDirectionalTemporalOutput[address + 2u * offset] = RoundSdfHistory(plane);
}
